<script>
function initTerminalPlan2D() {
    var viewport = document.getElementById('viewport');
    var canvas = document.getElementById('plan-canvas');
    var statusEl = document.getElementById('status');
    var bridge = document.getElementById('pick-bridge');
    var commandBridge = document.getElementById('scene-command-bridge');
    var errEl = document.getElementById('plan-error');
    var dataEl = document.getElementById('scene-data');

    function showError(text) {
        if (errEl) {
            errEl.style.display = 'block';
            errEl.textContent = text;
        }
    }

    if (!viewport || !canvas || typeof canvas.getContext !== 'function') {
        showError('Canvas 2D недоступен в этом клиенте.');
        return;
    }

    var ctx = canvas.getContext('2d');
    if (!ctx) {
        showError('Не удалось создать контекст Canvas 2D.');
        return;
    }

    var sceneData;
    try {
        var jsonText = dataEl ? (dataEl.textContent || dataEl.innerText || '') : '';
        sceneData = JSON.parse(jsonText);
    } catch (parseError) {
        showError('Ошибка JSON плана: ' + parseError.message);
        return;
    }

    if (!sceneData || sceneData.units !== 'm') {
        showError('Неподдерживаемые единицы или данные плана.');
        return;
    }

    function numberOr(value, fallback) {
        var n = Number(value);
        return isFinite(n) ? n : fallback;
    }

    var site = sceneData.site || {};
    site.width = numberOr(site.width, 720);
    site.length = numberOr(site.length, 620);
    site.quayWidth = Math.max(0, numberOr(site.quayWidth, 12));
    var rawShore = (site.shoreline instanceof Array) ? site.shoreline.slice() : [];
    site.shoreline = [];
    var si;
    for (si = 0; si < rawShore.length; si++) {
        var sp = rawShore[si] || {};
        site.shoreline.push({
            x: numberOr(sp.x, 0),
            y: numberOr(sp.y, 0)
        });
    }
    if (site.shoreline.length < 2) {
        var fallbackDepth = Math.max(0, numberOr(site.seaDepth, 100));
        site.shoreline = [
            { x: 0, y: fallbackDepth },
            { x: site.width, y: fallbackDepth }
        ];
    }

    var blocks = (sceneData.topology && sceneData.topology.blocks instanceof Array)
        ? sceneData.topology.blocks : [];
    var blockById = {};
    for (var bi = 0; bi < blocks.length; bi++) {
        blockById[String(blocks[bi].id)] = blocks[bi];
    }

    var cells = [];
    var cellById = {};
    var sourceCells = sceneData.cells instanceof Array ? sceneData.cells : [];
    for (var ci = 0; ci < sourceCells.length; ci++) {
        var sc = sourceCells[ci] || {};
        var cellId = String(sc.id == null ? '' : sc.id);
        if (!cellId || cellById[cellId]) {
            continue;
        }
        var cell = {
            id: cellId,
            kind: 'cell',
            blockId: String(sc.blockId || ''),
            row: numberOr(sc.row, 0),
            bay: numberOr(sc.bay, 0),
            count: Math.max(0, numberOr(sc.count, 0)),
            maxTier: Math.max(1, numberOr(sc.maxTier, 7)),
            x: numberOr(sc.x, 0),
            y: numberOr(sc.y, 0),
            rotation: numberOr(sc.rotation, 0)
        };
        cells.push(cell);
        cellById[cellId] = cell;
    }

    var objects = [];
    var objectById = {};
    var sourceObjects = sceneData.objects instanceof Array ? sceneData.objects : [];
    for (var oi = 0; oi < sourceObjects.length; oi++) {
        var so = sourceObjects[oi] || {};
        var oid = String(so.id == null ? '' : so.id);
        var kind = String(so.kind == null ? '' : so.kind);
        if (!oid || !kind || objectById[oid] || kind === 'cell') {
            continue;
        }
        var obj = {
            id: oid,
            kind: kind,
            name: String(so.name || oid),
            x: numberOr(so.x, 0),
            y: numberOr(so.y, 0),
            z: numberOr(so.z, 0),
            rotation: numberOr(so.rotation, 0),
            color: String(so.color || '#94a3b8'),
            size: so.size || {},
            load: String(so.load || 'none'),
            length: numberOr(so.length, numberOr(so.size && so.size.length, 0)),
            vesselId: String(so.vesselId || '')
        };
        objects.push(obj);
        objectById[oid] = obj;
    }

    var dpr = 1;
    var view = { scale: 1, offsetX: 40, offsetY: 40 };
    var selected = null;
    var needsDraw = true;
    var panning = false;
    var panLast = null;
    var pointerDown = null;
    var DRAG_THRESHOLD = 4;
    var buildMs = 0;

    var TIER_COLORS = [
        '#4a3f3f',
        '#f0a090',
        '#e87868',
        '#d45040',
        '#b83228',
        '#8f2018',
        '#6e1410',
        '#4a0c08'
    ];

    function colorForCount(count, maxTier) {
        if (count <= 0) {
            return TIER_COLORS[0];
        }
        var idx = Math.min(TIER_COLORS.length - 1, count);
        return TIER_COLORS[idx];
    }

    function worldToScreen(x, y) {
        return {
            x: view.offsetX + x * view.scale,
            y: view.offsetY + y * view.scale
        };
    }

    function screenToWorld(sx, sy) {
        return {
            x: (sx - view.offsetX) / view.scale,
            y: (sy - view.offsetY) / view.scale
        };
    }

    function resize() {
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        if (width < 1 || height < 1) {
            return;
        }
        dpr = (window.devicePixelRatio > 1.5) ? 1.5 : (window.devicePixelRatio || 1);
        if (dpr > 2) {
            dpr = 2;
        }
        canvas.width = Math.round(width * dpr);
        canvas.height = Math.round(height * dpr);
        canvas.style.width = width + 'px';
        canvas.style.height = height + 'px';
        needsDraw = true;
    }

    function resetView() {
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        var pad = 40;
        var sx = (width - pad * 2) / site.width;
        var sy = (height - pad * 2) / site.length;
        view.scale = Math.max(0.05, Math.min(sx, sy));
        view.offsetX = (width - site.width * view.scale) / 2;
        view.offsetY = (height - site.length * view.scale) / 2;
        needsDraw = true;
    }

    function shorelineCentroid() {
        var shore = site.shoreline;
        var i;
        var sx = 0;
        var sy = 0;
        for (i = 0; i < shore.length; i++) {
            sx += shore[i].x;
            sy += shore[i].y * 0.35;
        }
        return {
            x: sx / Math.max(1, shore.length),
            y: Math.max(24, sy / Math.max(1, shore.length))
        };
    }

    function beginSeaPolygon() {
        var shore = site.shoreline;
        var i;
        var p;
        ctx.beginPath();
        p = worldToScreen(0, 0);
        ctx.moveTo(p.x, p.y);
        p = worldToScreen(site.width, 0);
        ctx.lineTo(p.x, p.y);
        for (i = shore.length - 1; i >= 0; i--) {
            p = worldToScreen(shore[i].x, shore[i].y);
            ctx.lineTo(p.x, p.y);
        }
        ctx.closePath();
    }

    function beginShorePolyline() {
        var shore = site.shoreline;
        var i;
        var p;
        ctx.beginPath();
        for (i = 0; i < shore.length; i++) {
            p = worldToScreen(shore[i].x, shore[i].y);
            if (i === 0) {
                ctx.moveTo(p.x, p.y);
            } else {
                ctx.lineTo(p.x, p.y);
            }
        }
    }

    function drawRoundedSite() {
        var p0 = worldToScreen(0, 0);
        var sw = site.width * view.scale;
        var sh = site.length * view.scale;
        var shore = site.shoreline;
        var quayW = site.quayWidth;

        // суша
        ctx.fillStyle = '#2a3140';
        ctx.fillRect(p0.x, p0.y, sw, sh);

        // море — полигон: верх сцены + береговая линия
        beginSeaPolygon();
        ctx.fillStyle = '#1a6a9a';
        ctx.fill();

        ctx.save();
        beginSeaPolygon();
        ctx.clip();
        ctx.fillStyle = '#2280b5';
        ctx.fillRect(p0.x, p0.y, sw, Math.max(20, 42 * view.scale));
        // рябь внутри полигона
        ctx.strokeStyle = 'rgba(180, 230, 255, 0.22)';
        ctx.lineWidth = Math.max(1, view.scale * 0.35);
        var wy;
        var step = Math.max(6, 7 * view.scale);
        var maxSeaY = 0;
        var i;
        for (i = 0; i < shore.length; i++) {
            if (shore[i].y > maxSeaY) {
                maxSeaY = shore[i].y;
            }
        }
        for (wy = step * 0.6; wy < maxSeaY * view.scale - 2; wy += step) {
            ctx.beginPath();
            var wx;
            for (wx = 0; wx <= site.width; wx += 8) {
                var sx = p0.x + wx * view.scale;
                var amp = Math.sin((wx + wy) * 0.08) * 1.6 * view.scale;
                if (wx === 0) {
                    ctx.moveTo(sx, p0.y + wy + amp);
                } else {
                    ctx.lineTo(sx, p0.y + wy + amp);
                }
            }
            ctx.stroke();
        }
        ctx.restore();

        // причал вдоль полигона берега
        if (quayW > 0 && shore.length >= 2) {
            beginShorePolyline();
            ctx.strokeStyle = '#5c6578';
            ctx.lineWidth = Math.max(3, quayW * view.scale);
            ctx.lineJoin = 'round';
            ctx.lineCap = 'round';
            ctx.stroke();
            beginShorePolyline();
            ctx.strokeStyle = '#9aa3b5';
            ctx.lineWidth = Math.max(1, view.scale * 0.35);
            ctx.stroke();
            // кранцы на вершинах
            ctx.fillStyle = '#3d4658';
            for (i = 0; i < shore.length; i++) {
                var bp = worldToScreen(shore[i].x, shore[i].y);
                ctx.fillRect(
                    bp.x - Math.max(1.5, view.scale),
                    bp.y - Math.max(1.5, view.scale),
                    Math.max(3, 2.2 * view.scale),
                    Math.max(3, 2.4 * view.scale));
            }
        }

        if (view.scale > 0.7) {
            var c = shorelineCentroid();
            var cp = worldToScreen(c.x, c.y);
            ctx.fillStyle = 'rgba(220, 240, 255, 0.55)';
            ctx.font = Math.max(11, Math.min(16, 5.5 * view.scale)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('МОРЕ', cp.x, cp.y);
        }

        ctx.strokeStyle = '#465168';
        ctx.lineWidth = 1;
        ctx.strokeRect(p0.x, p0.y, sw, sh);
    }

    function drawCell(cell) {
        var block = blockById[cell.blockId];
        var w = block ? numberOr(block.bayPitch, 12.5) * 0.92 : 11;
        var h = block ? numberOr(block.rowPitch, 2.75) * 0.85 : 2.3;
        var ang = cell.rotation * Math.PI / 180;
        var p = worldToScreen(cell.x, cell.y);
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(ang);
        var sw = w * view.scale;
        var sh = h * view.scale;
        ctx.fillStyle = colorForCount(cell.count, cell.maxTier);
        ctx.fillRect(-sw / 2, -sh / 2, sw, sh);
        if (selected && selected.id === cell.id) {
            ctx.strokeStyle = '#ffcc33';
            ctx.lineWidth = 2;
            ctx.strokeRect(-sw / 2, -sh / 2, sw, sh);
        } else if (view.scale > 2.5) {
            ctx.strokeStyle = 'rgba(0,0,0,0.25)';
            ctx.lineWidth = 1;
            ctx.strokeRect(-sw / 2, -sh / 2, sw, sh);
        }
        if (view.scale > 4 && cell.count > 0) {
            ctx.fillStyle = '#fff';
            ctx.font = Math.max(9, Math.min(14, 5 * view.scale)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(String(cell.count), 0, 0);
        }
        ctx.restore();
    }

    function drawBlockOutline(block) {
        var rows = Math.max(1, numberOr(block.rows, 1));
        var bays = Math.max(1, numberOr(block.bays, 1));
        var rowPitch = numberOr(block.rowPitch, 2.75);
        var bayPitch = numberOr(block.bayPitch, 12.5);
        var ang = numberOr(block.rotationDeg, 0) * Math.PI / 180;
        var cosA = Math.cos(ang);
        var sinA = Math.sin(ang);
        var corners = [
            { lx: -bayPitch * 0.5, ly: -rowPitch * 0.5 },
            { lx: (bays - 0.5) * bayPitch, ly: -rowPitch * 0.5 },
            { lx: (bays - 0.5) * bayPitch, ly: (rows - 0.5) * rowPitch },
            { lx: -bayPitch * 0.5, ly: (rows - 0.5) * rowPitch }
        ];
        ctx.beginPath();
        for (var i = 0; i < corners.length; i++) {
            var wx = block.originX + corners[i].lx * cosA - corners[i].ly * sinA;
            var wy = block.originY + corners[i].lx * sinA + corners[i].ly * cosA;
            var p = worldToScreen(wx, wy);
            if (i === 0) {
                ctx.moveTo(p.x, p.y);
            } else {
                ctx.lineTo(p.x, p.y);
            }
        }
        ctx.closePath();
        ctx.strokeStyle = '#8fa0b8';
        ctx.lineWidth = 1.5;
        ctx.stroke();
    }

    function strokeFillRect(x, y, w, h) {
        ctx.fillRect(x, y, w, h);
        ctx.strokeRect(x, y, w, h);
    }

    function drawWheels(axles, halfW, wheelR) {
        var i;
        ctx.fillStyle = '#1a1a1a';
        for (i = 0; i < axles.length; i++) {
            ctx.beginPath();
            ctx.arc(axles[i], -halfW, wheelR, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.arc(axles[i], halfW, wheelR, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    function vesselLength(obj) {
        var size = obj.size || {};
        return numberOr(obj.length, numberOr(size.length, 120));
    }

    function vesselBeam(obj) {
        var size = obj.size || {};
        return numberOr(size.beam, numberOr(size.width, 28));
    }

    function equipmentBounds(obj) {
        var kind = obj.kind || '';
        var size = obj.size || {};
        if (kind === 'warehouse') {
            return {
                hw: numberOr(size.length, 30) * 0.5,
                hh: numberOr(size.width, 18) * 0.5
            };
        }
        if (kind === 'vessel') {
            return {
                hw: vesselLength(obj) * 0.52,
                hh: vesselBeam(obj) * 0.55
            };
        }
        if (kind === 'craneSts') {
            return {
                hw: numberOr(size.length, 14) * 0.7,
                hh: (numberOr(size.outreach, 42) + numberOr(size.gauge, 18) * 0.5) * 0.55
            };
        }
        if (kind === 'craneRtg') {
            return {
                hw: numberOr(size.length, 22) * 0.55,
                hh: numberOr(size.span, 32) * 0.55
            };
        }
        if (kind === 'craneReachStacker') {
            return { hw: 14, hh: 5 };
        }
        if (kind === 'craneMobile') {
            return { hw: 12, hh: 6 };
        }
        if (kind === 'railway') {
            return { hw: numberOr(size.length, 100) * 0.5, hh: 4 };
        }
        if (kind === 'containerTruck') {
            return { hw: 11, hh: 3.2 };
        }
        if (kind === 'dieselLocomotive') {
            return { hw: 9, hh: 2.2 };
        }
        if (kind.indexOf('wagon') === 0) {
            return {
                hw: numberOr(size.length, 14) * 0.55,
                hh: numberOr(size.width, 3.1) * 0.7
            };
        }
        return { hw: 6, hh: 6 };
    }

    function drawVesselIcon(obj, s) {
        var len = vesselLength(obj) * s;
        var beam = vesselBeam(obj) * s;
        var color = obj.color || '#d7dee8';
        var x0 = -len / 2;
        var y0 = -beam / 2;
        ctx.lineJoin = 'round';
        ctx.strokeStyle = '#1f2937';
        ctx.lineWidth = Math.max(1.2, s * 0.12);

        // корпус
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.moveTo(x0, y0 + beam * 0.18);
        ctx.lineTo(x0 + len * 0.12, y0);
        ctx.lineTo(x0 + len * 0.78, y0);
        ctx.lineTo(x0 + len * 0.92, y0 + beam * 0.22);
        ctx.lineTo(x0 + len, y0 + beam * 0.5); // нос
        ctx.lineTo(x0 + len * 0.92, y0 + beam * 0.78);
        ctx.lineTo(x0 + len * 0.78, y0 + beam);
        ctx.lineTo(x0 + len * 0.12, y0 + beam);
        ctx.lineTo(x0, y0 + beam * 0.82);
        ctx.closePath();
        ctx.fill();
        ctx.stroke();

        // палуба / грузовые трюмы
        ctx.fillStyle = '#64748b';
        strokeFillRect(x0 + len * 0.14, y0 + beam * 0.22, len * 0.58, beam * 0.56);
        ctx.fillStyle = 'rgba(30, 58, 95, 0.35)';
        var hi;
        var hatchW = len * 0.12;
        for (hi = 0; hi < 4; hi++) {
            ctx.fillRect(
                x0 + len * 0.16 + hi * (hatchW + len * 0.02),
                y0 + beam * 0.28,
                hatchW,
                beam * 0.44);
        }

        // надстройка / мостик у кормы
        ctx.fillStyle = '#e2e8f0';
        strokeFillRect(x0 + len * 0.04, y0 + beam * 0.18, len * 0.12, beam * 0.64);
        ctx.fillStyle = '#1e3a5f';
        strokeFillRect(x0 + len * 0.06, y0 + beam * 0.28, len * 0.08, beam * 0.44);
        ctx.fillStyle = '#93c5fd';
        ctx.fillRect(x0 + len * 0.075, y0 + beam * 0.34, len * 0.05, beam * 0.18);

        // ватерлиния
        ctx.strokeStyle = '#334155';
        ctx.lineWidth = Math.max(1, s * 0.1);
        ctx.beginPath();
        ctx.moveTo(x0 + len * 0.08, 0);
        ctx.lineTo(x0 + len * 0.88, 0);
        ctx.stroke();

        if (view.scale > 0.9) {
            ctx.fillStyle = '#0f172a';
            ctx.font = 'bold ' + Math.max(9, Math.min(13, 3.2 * s)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(obj.name || 'Судно', x0 + len * 0.45, 0);
            if (view.scale > 1.4) {
                ctx.font = Math.max(8, Math.min(11, 2.4 * s)) + 'px Segoe UI, sans-serif';
                ctx.fillStyle = '#334155';
                ctx.fillText('L=' + Math.round(vesselLength(obj)) + ' м', x0 + len * 0.45, beam * 0.28);
            }
        }
    }

    function drawStsIcon(obj, s) {
        var gauge = numberOr(obj.size.gauge, 18) * s;
        var outreach = numberOr(obj.size.outreach, 42) * s;
        var len = numberOr(obj.size.length, 14) * s;
        var color = obj.color || '#f0c000';
        var leg = Math.max(2.2, 1.4 * s);
        // локально: +Y к суше, −Y к морю (вылет стрелы)
        var landY = gauge * 0.35;
        var seaY = -outreach;

        ctx.strokeStyle = '#111';
        ctx.lineWidth = 1;

        // рельсовые тележки на причале
        ctx.fillStyle = '#374151';
        strokeFillRect(-len / 2, landY - leg * 0.2, len, leg);
        strokeFillRect(-len / 2, landY + gauge * 0.55, len, leg * 0.85);

        // портал
        ctx.fillStyle = color;
        strokeFillRect(-len / 2, landY - leg * 0.2, leg, gauge * 0.85 + leg);
        strokeFillRect(len / 2 - leg, landY - leg * 0.2, leg, gauge * 0.85 + leg);
        // поперечная балка портала
        var beamH = Math.max(2, 1.3 * s);
        strokeFillRect(-len / 2, landY + gauge * 0.15, len, beamH);

        // стрела к морю
        ctx.fillStyle = color;
        strokeFillRect(-leg * 0.55, seaY, leg * 1.1, landY - seaY + beamH);
        // троллей + спредер
        ctx.fillStyle = '#1f2937';
        var trolleyY = seaY + outreach * 0.45;
        strokeFillRect(-2.4 * s, trolleyY - 1.2 * s, 4.8 * s, 2.4 * s);
        ctx.strokeStyle = color;
        ctx.lineWidth = Math.max(1.5, s * 0.28);
        ctx.strokeRect(-3.2 * s, trolleyY + 1.4 * s, 6.4 * s, Math.max(2.5, 2 * s));

        // оттяжка
        ctx.strokeStyle = '#92400e';
        ctx.lineWidth = Math.max(1, s * 0.15);
        ctx.beginPath();
        ctx.moveTo(0, landY + gauge * 0.15);
        ctx.lineTo(0, seaY + 2 * s);
        ctx.stroke();

        if (view.scale > 1.3) {
            ctx.fillStyle = '#1f2937';
            ctx.font = Math.max(8, Math.min(11, 2.6 * s)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(obj.name || 'STS', 0, landY + gauge * 0.45);
        }
    }

    function drawWarehouseIcon(obj, s) {
        var wl = numberOr(obj.size.length, 30) * s;
        var ww = numberOr(obj.size.width, 18) * s;
        var x0 = -wl / 2;
        var y0 = -ww / 2;
        ctx.globalAlpha = 0.92;
        ctx.fillStyle = obj.color || '#64748b';
        ctx.strokeStyle = '#1f2937';
        ctx.lineWidth = Math.max(1, s * 0.08);
        strokeFillRect(x0, y0, wl, ww);
        // конёк крыши
        ctx.strokeStyle = '#94a3b8';
        ctx.beginPath();
        ctx.moveTo(x0 + wl * 0.08, 0);
        ctx.lineTo(x0 + wl * 0.92, 0);
        ctx.stroke();
        // ворота / рампы по длинной стороне
        var doors = Math.max(2, Math.min(6, Math.round(wl / (18 * s))));
        var i;
        var doorW = Math.max(4, 4.5 * s);
        var gap = wl / (doors + 1);
        ctx.fillStyle = '#334155';
        for (i = 1; i <= doors; i++) {
            var dx = x0 + gap * i - doorW / 2;
            ctx.fillRect(dx, y0 + ww - Math.max(3, 2.2 * s), doorW, Math.max(3, 2.2 * s));
        }
        // световые фонари на крыше
        ctx.fillStyle = 'rgba(226, 232, 240, 0.35)';
        for (i = 0; i < 3; i++) {
            ctx.fillRect(x0 + wl * (0.2 + i * 0.25), y0 + ww * 0.22, wl * 0.12, ww * 0.18);
        }
        ctx.globalAlpha = 1;
        if (view.scale > 1.2) {
            ctx.fillStyle = '#e2e8f0';
            ctx.font = Math.max(9, Math.min(13, 4.5 * s)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(obj.name || 'Склад', 0, -ww * 0.12);
        }
    }

    function applyDprTransform() {
        try {
            if (typeof ctx.setTransform === 'function') {
                ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
                return;
            }
        } catch (ignoreSet) {}
        try {
            if (typeof ctx.scale === 'function') {
                ctx.scale(dpr, dpr);
            }
        } catch (ignoreScale) {}
    }

    function draw() {
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        applyDprTransform();
        ctx.clearRect(0, 0, width, height);
        drawRoundedSite();

        var i;
        for (i = 0; i < cells.length; i++) {
            drawCell(cells[i]);
        }
        for (i = 0; i < blocks.length; i++) {
            drawBlockOutline(blocks[i]);
        }
        // суда под STS и остальной техникой
        for (i = 0; i < objects.length; i++) {
            if (objects[i].kind === 'vessel') {
                drawEquipment(objects[i]);
            }
        }
        for (i = 0; i < objects.length; i++) {
            if (objects[i].kind !== 'vessel') {
                drawEquipment(objects[i]);
            }
        }
        needsDraw = false;
        updateStatus();
    }

    function updateStatus() {
        if (!statusEl) {
            return;
        }
        var text = 'Ячеек: ' + cells.length + ' · объектов: ' + objects.length +
            ' · масштаб: ' + view.scale.toFixed(2) + ' px/м';
        if (selected) {
            if (selected.kind === 'cell') {
                text += ' · ' + selected.id + ' count=' + selected.count +
                    '/' + selected.maxTier;
            } else {
                text += ' · ' + selected.name + ' [' + selected.kind + ']';
            }
        }
        statusEl.textContent = text;
    }

    function hitTest(sx, sy) {
        var world = screenToWorld(sx, sy);
        var i;
        for (i = objects.length - 1; i >= 0; i--) {
            var obj = objects[i];
            var b = equipmentBounds(obj);
            var ang = (obj.rotation || 0) * Math.PI / 180;
            var cosA = Math.cos(ang);
            var sinA = Math.sin(ang);
            var lx = (world.x - obj.x) * cosA + (world.y - obj.y) * sinA;
            var ly = -(world.x - obj.x) * sinA + (world.y - obj.y) * cosA;
            var pad = (obj.kind === 'craneReachStacker' || obj.kind === 'craneMobile') ? 2 : 0.5;
            if (Math.abs(lx) <= b.hw + pad && Math.abs(ly) <= b.hh + pad) {
                return obj;
            }
        }
        var best = null;
        var bestDist = 1e9;
        for (i = 0; i < cells.length; i++) {
            var cell = cells[i];
            var block = blockById[cell.blockId];
            var hw = (block ? numberOr(block.bayPitch, 12.5) : 12) * 0.5;
            var hh = (block ? numberOr(block.rowPitch, 2.75) : 2.75) * 0.5;
            var cAng = cell.rotation * Math.PI / 180;
            var cCos = Math.cos(cAng);
            var cSin = Math.sin(cAng);
            var clx = (world.x - cell.x) * cCos + (world.y - cell.y) * cSin;
            var cly = -(world.x - cell.x) * cSin + (world.y - cell.y) * cCos;
            if (Math.abs(clx) <= hw && Math.abs(cly) <= hh) {
                var d = clx * clx + cly * cly;
                if (d < bestDist) {
                    bestDist = d;
                    best = cell;
                }
            }
        }
        return best;
    }

    function notify1C(target) {
        if (!bridge || !target) {
            return;
        }
        bridge.setAttribute('href', '#pick:' + encodeURIComponent(target.id));
        bridge.setAttribute('data-event', 'pick');
        bridge.setAttribute('data-id', target.id);
        bridge.setAttribute('data-kind', target.kind);
        bridge.setAttribute('data-x', JSON.stringify(target.x));
        bridge.setAttribute('data-y', JSON.stringify(target.y));
        bridge.setAttribute('data-z', JSON.stringify(target.z || 0));
        if (target.kind === 'cell') {
            bridge.setAttribute('data-count', JSON.stringify(target.count));
            bridge.setAttribute('data-block', target.blockId);
            bridge.setAttribute('data-row', JSON.stringify(target.row));
            bridge.setAttribute('data-bay', JSON.stringify(target.bay));
        }
        if (typeof bridge.click === 'function') {
            bridge.click();
        } else {
            var evt = document.createEvent('MouseEvents');
            evt.initEvent('click', true, true);
            bridge.dispatchEvent(evt);
        }
    }

    function selectTarget(target) {
        selected = target || null;
        needsDraw = true;
        if (target) {
            notify1C(target);
        }
        updateStatus();
    }

    function focusObject(id) {
        id = String(id == null ? '' : id);
        var target = cellById[id] || objectById[id];
        if (!target) {
            throw new Error('Объект с id "' + id + '" не найден.');
        }
        selectTarget(target);
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        view.offsetX = width / 2 - target.x * view.scale;
        view.offsetY = height / 2 - target.y * view.scale;
        needsDraw = true;
        return target;
    }

    function moveObject(id, x, y, z) {
        var obj = objectById[String(id)];
        if (!obj) {
            throw new Error('Объект с id "' + id + '" не найден.');
        }
        obj.x = Number(x);
        obj.y = Number(y);
        if (isFinite(Number(z))) {
            obj.z = Number(z);
        }
        needsDraw = true;
        return obj;
    }

    function upsertCell(data) {
        data = data || {};
        var id = String(data.id == null ? '' : data.id);
        if (!id) {
            throw new Error('cellUpsert без id');
        }
        var cell = cellById[id];
        if (!cell) {
            cell = {
                id: id,
                kind: 'cell',
                blockId: String(data.blockId || ''),
                row: numberOr(data.row, 0),
                bay: numberOr(data.bay, 0),
                count: 0,
                maxTier: numberOr(data.maxTier, 7),
                x: numberOr(data.x, 0),
                y: numberOr(data.y, 0),
                rotation: numberOr(data.rotation, 0)
            };
            cells.push(cell);
            cellById[id] = cell;
        }
        if (data.count != null) {
            cell.count = Math.max(0, numberOr(data.count, cell.count));
        }
        if (data.maxTier != null) {
            cell.maxTier = Math.max(1, numberOr(data.maxTier, cell.maxTier));
        }
        if (data.x != null) {
            cell.x = numberOr(data.x, cell.x);
        }
        if (data.y != null) {
            cell.y = numberOr(data.y, cell.y);
        }
        needsDraw = true;
        return cell;
    }

    function applyDelta(delta) {
        delta = delta || {};
        var started = (typeof performance !== 'undefined' && performance.now)
            ? performance.now() : Date.now();
        var moves = delta.moves instanceof Array ? delta.moves : [];
        var cellUpserts = delta.cellUpserts instanceof Array ? delta.cellUpserts : [];
        var mi;
        for (mi = 0; mi < moves.length; mi++) {
            var m = moves[mi] || {};
            moveObject(m.id, m.x, m.y, m.z);
        }
        for (mi = 0; mi < cellUpserts.length; mi++) {
            upsertCell(cellUpserts[mi]);
        }
        var ended = (typeof performance !== 'undefined' && performance.now)
            ? performance.now() : Date.now();
        needsDraw = true;
        return {
            moves: moves.length,
            cellUpserts: cellUpserts.length,
            ms: Math.round(ended - started)
        };
    }

    function applyCommand(command) {
        command = command || {};
        if (command.op === 'move') {
            return moveObject(command.id, command.x, command.y, command.z);
        }
        if (command.op === 'applyDelta') {
            return applyDelta(command.delta);
        }
        if (command.op === 'focus') {
            return focusObject(command.id);
        }
        if (command.op === 'upsertCell') {
            return upsertCell(command.cell);
        }
        throw new Error('Неизвестная операция "' + String(command.op || '') + '".');
    }

    function processBridgeCommand(event) {
        if (event) {
            if (event.preventDefault) {
                event.preventDefault();
            }
            if (event.stopPropagation) {
                event.stopPropagation();
            }
        }
        if (!commandBridge) {
            return false;
        }
        var sequence = commandBridge.getAttribute('data-seq') || '';
        if (!sequence || sequence === processBridgeCommand.lastSequence) {
            return false;
        }
        processBridgeCommand.lastSequence = sequence;
        try {
            applyCommand(JSON.parse(commandBridge.getAttribute('data-payload') || ''));
            commandBridge.setAttribute('data-error', '');
        } catch (commandError) {
            commandBridge.setAttribute('data-error', commandError.message || String(commandError));
        }
        commandBridge.setAttribute('data-ack', sequence);
        return false;
    }
    processBridgeCommand.lastSequence = '';

    window.terminalPlan2d = {
        applyDelta: applyDelta,
        applyCommand: applyCommand,
        focusObject: focusObject,
        getCell: function (id) { return cellById[String(id)] || null; },
        getObject: function (id) { return objectById[String(id)] || null; }
    };

    if (commandBridge) {
        if (commandBridge.addEventListener) {
            commandBridge.addEventListener('click', processBridgeCommand, false);
        } else if (commandBridge.attachEvent) {
            commandBridge.attachEvent('onclick', processBridgeCommand);
        } else {
            commandBridge.onclick = processBridgeCommand;
        }
        commandBridge.setAttribute('data-ready', 'true');
    }

    function bind(el, eventName, handler, useCapture) {
        if (!el) {
            return;
        }
        if (el.addEventListener) {
            try {
                el.addEventListener(eventName, handler, !!useCapture);
            } catch (ignoreBind) {
                el.addEventListener(eventName, handler);
            }
        } else if (el.attachEvent) {
            el.attachEvent('on' + eventName, handler);
        }
    }

    function onWheel(e) {
        if (!e) { e = window.event; }
        if (e.preventDefault) { e.preventDefault(); }
        var rect = canvas.getBoundingClientRect();
        var sx = e.clientX - rect.left;
        var sy = e.clientY - rect.top;
        var before = screenToWorld(sx, sy);
        var delta = e.deltaY != null ? e.deltaY : (e.wheelDelta ? -e.wheelDelta : 0);
        var factor = delta > 0 ? 0.9 : 1.1;
        view.scale = Math.max(0.05, Math.min(20, view.scale * factor));
        view.offsetX = sx - before.x * view.scale;
        view.offsetY = sy - before.y * view.scale;
        needsDraw = true;
        return false;
    }

    function bootDraw() {
        resize();
        resetView();
        needsDraw = true;
        draw();
        if (statusEl) {
            statusEl.textContent = 'Ячеек: ' + cells.length +
                ' · объектов: ' + objects.length +
                ' · масштаб ' + view.scale.toFixed(2);
        }
    }

    // Сначала рисуем — до API, которые могут упасть в WebKit 1С.
    bootDraw();
    setTimeout(bootDraw, 50);
    setTimeout(bootDraw, 250);

    bind(canvas, 'mousedown', function (e) {
        if (!e) { e = window.event; }
        if (e.button !== 0) { return; }
        panning = true;
        panLast = { x: e.clientX, y: e.clientY };
        pointerDown = { x: e.clientX, y: e.clientY, moved: false };
        viewport.className = (viewport.className || '').replace(/\bpanning\b/g, '') + ' panning';
        if (e.preventDefault) { e.preventDefault(); }
    });

    bind(window, 'mousemove', function (e) {
        if (!e) { e = window.event; }
        if (!panning || !panLast) { return; }
        var dx = e.clientX - panLast.x;
        var dy = e.clientY - panLast.y;
        panLast = { x: e.clientX, y: e.clientY };
        if (pointerDown) {
            var tx = e.clientX - pointerDown.x;
            var ty = e.clientY - pointerDown.y;
            if (tx * tx + ty * ty > DRAG_THRESHOLD * DRAG_THRESHOLD) {
                pointerDown.moved = true;
            }
        }
        view.offsetX += dx;
        view.offsetY += dy;
        needsDraw = true;
    });

    bind(window, 'mouseup', function (e) {
        if (!e) { e = window.event; }
        if (e.button !== 0) { return; }
        var wasClick = pointerDown && !pointerDown.moved;
        var clickX = e.clientX;
        var clickY = e.clientY;
        panning = false;
        panLast = null;
        pointerDown = null;
        viewport.className = (viewport.className || '').replace(/\bpanning\b/g, '').replace(/\s+/g, ' ');
        if (wasClick) {
            var rect = canvas.getBoundingClientRect();
            selectTarget(hitTest(clickX - rect.left, clickY - rect.top));
        }
    });

    bind(canvas, 'mousewheel', onWheel);
    bind(canvas, 'wheel', onWheel);
    canvas.onmousewheel = onWheel;

    bind(document.getElementById('btn-reset'), 'click', function () {
        resetView();
        needsDraw = true;
        draw();
    });

    bind(window, 'resize', function () {
        resize();
        needsDraw = true;
    });
    try {
        if (typeof ResizeObserver !== 'undefined') {
            new ResizeObserver(function () {
                resize();
                needsDraw = true;
            }).observe(viewport);
        }
    } catch (ignoreRo) {}

    var raf = window.requestAnimationFrame || window.webkitRequestAnimationFrame ||
        function (cb) { return setTimeout(function () { cb(); }, 33); };

    function loop() {
        if (needsDraw) {
            draw();
        }
        raf(loop);
    }
    raf(loop);
}

function startTerminalPlan2D() {
    try {
        initTerminalPlan2D();
    } catch (bootError) {
        var errNode = document.getElementById('plan-error');
        var st = document.getElementById('status');
        var msg = 'Ошибка плана: ' + (bootError && bootError.message
            ? bootError.message : String(bootError));
        if (errNode) {
            errNode.style.display = 'block';
            errNode.textContent = msg;
        }
        if (st) {
            st.textContent = msg;
        }
    }
}

if (document.readyState === 'loading') {
    if (document.addEventListener) {
        document.addEventListener('DOMContentLoaded', startTerminalPlan2D, false);
    } else {
        setTimeout(startTerminalPlan2D, 0);
    }
} else {
    startTerminalPlan2D();
}
