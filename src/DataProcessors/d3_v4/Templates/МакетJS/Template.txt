<script>
function initTerminalPlan2D() {
    var viewport = document.getElementById('viewport');
    var canvas = document.getElementById('plan-canvas');
    var bridge = document.getElementById('pick-bridge');
    var commandBridge = document.getElementById('scene-command-bridge');
    var errEl = document.getElementById('plan-error');
    var dataEl = document.getElementById('scene-data');

    function showError(text) {
        if (errEl) {
            errEl.style.display = 'block';
            errEl.textContent = text;
        }
    }

    if (!viewport || !canvas || typeof canvas.getContext !== 'function') {
        showError('Canvas 2D недоступен в этом клиенте.');
        return;
    }

    var ctx = canvas.getContext('2d');
    if (!ctx) {
        showError('Не удалось создать контекст Canvas 2D.');
        return;
    }

    var sceneData;
    try {
        var jsonText = dataEl ? (dataEl.textContent || dataEl.innerText || '') : '';
        sceneData = JSON.parse(jsonText);
    } catch (parseError) {
        showError('Ошибка JSON плана: ' + parseError.message);
        return;
    }

    if (!sceneData || sceneData.units !== 'm') {
        showError('Неподдерживаемые единицы или данные плана.');
        return;
    }

    function numberOr(value, fallback) {
        var n = Number(value);
        return isFinite(n) ? n : fallback;
    }

    function plannedContainerCounts(value) {
        var src = (value && typeof value === 'object') ? value : {};
        return {
            ft10: Math.max(0, numberOr(src.ft10, 0)),
            ft20: Math.max(0, numberOr(src.ft20, 0)),
            ft40: Math.max(0, numberOr(src.ft40, 0))
        };
    }

    var site = sceneData.site || {};
    site.minX = numberOr(site.minX, 0);
    site.minY = numberOr(site.minY, 0);
    site.width = numberOr(site.width, 720);
    site.length = numberOr(site.length, 620);
    site.maxX = numberOr(site.maxX, site.minX + site.width);
    site.maxY = numberOr(site.maxY, site.minY + site.length);
    site.quayWidth = Math.max(0, numberOr(site.quayWidth, 4));
    function normalizePointRing(raw) {
        var out = [];
        var i;
        if (!(raw instanceof Array)) {
            return out;
        }
        for (i = 0; i < raw.length; i++) {
            var pt = raw[i] || {};
            out.push({
                x: numberOr(pt.x, 0),
                y: numberOr(pt.y, 0)
            });
        }
        return out;
    }

    var rawShore = (site.shoreline instanceof Array) ? site.shoreline.slice() : [];
    site.shoreline = normalizePointRing(rawShore);
    if (site.shoreline.length < 2) {
        var fallbackDepth = Math.max(0, numberOr(site.seaDepth, 100));
        site.shoreline = [
            { x: 0, y: fallbackDepth },
            { x: site.width, y: fallbackDepth }
        ];
    }

    // site.water: одно кольцо [{x,y},…] или массив колец [[{x,y},…], …]
    site.waterRings = [];
    var rawWater = site.water;
    if (rawWater instanceof Array && rawWater.length > 0) {
        if (rawWater[0] && rawWater[0].x != null) {
            site.waterRings.push(normalizePointRing(rawWater));
        } else {
            var wi;
            for (wi = 0; wi < rawWater.length; wi++) {
                var ring = normalizePointRing(rawWater[wi]);
                if (ring.length >= 3) {
                    site.waterRings.push(ring);
                }
            }
        }
    }

    var containerSites = (sceneData.topology && sceneData.topology.sites instanceof Array)
        ? sceneData.topology.sites : [];
    var zones = (sceneData.topology && sceneData.topology.zones instanceof Array)
        ? sceneData.topology.zones : [];
    var infrastructure = (sceneData.topology && sceneData.topology.infrastructure instanceof Array)
        ? sceneData.topology.infrastructure : [];
    var buildings = [];
    var buildingById = {};
    var rawBuildings = (sceneData.topology && sceneData.topology.buildings instanceof Array)
        ? sceneData.topology.buildings : [];
    for (var initialBuildingIndex = 0; initialBuildingIndex < rawBuildings.length; initialBuildingIndex++) {
        var rawBuilding = rawBuildings[initialBuildingIndex] || {};
        var buildingPolygon = normalizePointRing(rawBuilding.polygon);
        if (buildingPolygon.length < 3) {
            continue;
        }
        var buildingId = String(rawBuilding.id || '');
        if (!buildingId || buildingById[buildingId]) {
            continue;
        }
        var buildingCenter = ringCentroid(buildingPolygon);
        var building = {
            id: buildingId,
            kind: 'building',
            name: String(rawBuilding.name || rawBuilding.id || ''),
            color: String(rawBuilding.color || '#64748b'),
            x: numberOr(rawBuilding.x, buildingCenter.x),
            y: numberOr(rawBuilding.y, buildingCenter.y),
            polygon: buildingPolygon
        };
        buildings.push(building);
        buildingById[buildingId] = building;
    }
    function normalizeRoadList(rawList) {
        var result = [];
        if (!(rawList instanceof Array)) {
            return result;
        }
        for (var roadIndex = 0; roadIndex < rawList.length; roadIndex++) {
            var rawRoad = rawList[roadIndex] || {};
            var roadPoints = normalizePointRing(rawRoad.points);
            var roadId = String(rawRoad.id || '');
            var widthM = numberOr(rawRoad.widthM, 0);
            if (roadPoints.length < 2 || !roadId || widthM <= 0) {
                continue;
            }
            result.push({
                id: roadId,
                name: String(rawRoad.name || roadId),
                widthM: widthM,
                color: String(rawRoad.color || '#6b7280'),
                points: roadPoints
            });
        }
        return result;
    }
    var roads = normalizeRoadList(
        (sceneData.topology && sceneData.topology.roads instanceof Array)
            ? sceneData.topology.roads : []);
    var portBoundary = normalizeRoadList(
        (sceneData.topology && sceneData.topology.portBoundary instanceof Array)
            ? sceneData.topology.portBoundary : []);
    function normalizeRailwayList(rawList, requireId) {
        var result = [];
        if (!(rawList instanceof Array)) {
            return result;
        }
        for (var railwayIndex = 0; railwayIndex < rawList.length; railwayIndex++) {
            var rawRailway = rawList[railwayIndex] || {};
            var railwayPoints = normalizePointRing(rawRailway.points);
            var railwayId = String(rawRailway.id || '');
            if (railwayPoints.length < 2 || (requireId && !railwayId)) {
                continue;
            }
            result.push({
                id: railwayId,
                name: String(rawRailway.name || railwayId),
                gaugeM: Math.max(0.1, numberOr(rawRailway.gaugeM, 1.52)),
                color: String(rawRailway.color || '#6b7280'),
                points: railwayPoints
            });
        }
        return result;
    }
    var rawRailways = (sceneData.topology && sceneData.topology.railways)
        ? sceneData.topology.railways : {};
    var visualRailways = normalizeRailwayList(rawRailways.visual, false);
    var railwayBranches = normalizeRailwayList(rawRailways.branches, true);
    var drawRailwayBranches = Boolean(sceneData.drawRailwayBranches);
    var cells = (sceneData.cells instanceof Array) ? sceneData.cells.slice() : [];
    var cellById = {};
    for (var initialCellIndex = 0; initialCellIndex < cells.length; initialCellIndex++) {
        var initialCell = cells[initialCellIndex] || {};
        initialCell.id = String(initialCell.id || '');
        initialCell.kind = 'cell';
        initialCell.x = numberOr(initialCell.x, 0);
        initialCell.y = numberOr(initialCell.y, 0);
        initialCell.rotation = numberOr(initialCell.rotation, 0);
        initialCell.zoneKind = String(initialCell.zoneKind || 'unknown');
        initialCell.labelRotationDeg = numberOr(initialCell.labelRotationDeg, 0) === 180 ? 180 : 0;
        initialCell.length = Math.max(0.1, numberOr(initialCell.length, 6.1));
        initialCell.width = Math.max(0.1, numberOr(initialCell.width, 2.5));
        initialCell.count = 0;
        initialCell.maxTier = Math.max(1, numberOr(initialCell.maxTier, 1));
        cellById[initialCell.id] = initialCell;
    }
    var placements = [];
    var placementById = {};
    var topPlacementList = [];
    // cellId|tier -> placementId
    var occupancy = {};

    function occupancyKey(cellId, tier) {
        return String(cellId) + '|' + String(tier);
    }

    function recomputeCellCounts() {
        var ci;
        for (ci = 0; ci < cells.length; ci++) {
            cells[ci].count = 0;
        }
        var pi;
        for (pi = 0; pi < placements.length; pi++) {
            var pl = placements[pi];
            var ids = pl.cellIds || [];
            var ii;
            for (ii = 0; ii < ids.length; ii++) {
                var cell = cellById[String(ids[ii])];
                if (cell && pl.tier > cell.count) {
                    cell.count = pl.tier;
                }
            }
        }
        rebuildTopPlacementList();
    }

    function rebuildTopPlacementList() {
        topPlacementList = [];
        var seen = {};
        var pi;
        for (pi = 0; pi < placements.length; pi++) {
            var pl = placements[pi];
            var cell = cellById[pl.cellIds[0]];
            if (!cell || cell.count !== pl.tier) {
                continue;
            }
            // верхнее, если на якорной ячейке это max tier
            if (seen[pl.id]) {
                continue;
            }
            seen[pl.id] = true;
            topPlacementList.push(pl);
        }
    }

    function normalizePlacement(raw) {
        raw = raw || {};
        var blockId = String(raw.blockId || '');
        var row = Math.max(1, numberOr(raw.row, 1));
        var bay = Math.max(1, numberOr(raw.bay, 1));
        var tier = Math.max(1, numberOr(raw.tier, 1));
        var lengthFt = numberOr(raw.lengthFt, 20) >= 40 ? 40 : 20;
        var span = lengthFt === 40 ? 2 : 1;
        var id = String(raw.id == null || raw.id === ''
            ? ('P-' + blockId + '-' + row + '-' + bay + '-T' + tier + '-' + lengthFt)
            : raw.id);
        var cellIds = [];
        if (raw.cellIds instanceof Array && raw.cellIds.length) {
            var cii;
            for (cii = 0; cii < raw.cellIds.length; cii++) {
                cellIds.push(String(raw.cellIds[cii]));
            }
        } else {
            throw new Error('Размещение "' + id + '" не содержит cellIds.');
        }
        var anchor = cellById[cellIds[0]];
        if (!anchor) {
            throw new Error('Якорная ячейка "' + cellIds[0] + '" не найдена.');
        }
        var pos = anchor;
        if (span === 2 && cellById[cellIds[1]]) {
            pos = {
                x: (cellById[cellIds[0]].x + cellById[cellIds[1]].x) / 2,
                y: (cellById[cellIds[0]].y + cellById[cellIds[1]].y) / 2,
                rotation: cellById[cellIds[0]].rotation
            };
        }
        return {
            id: id,
            kind: 'placement',
            name: id,
            blockId: blockId,
            zoneId: String(raw.zoneId || anchor.zoneId || ''),
            zoneKind: String(raw.zoneKind || anchor.zoneKind || 'unknown'),
            row: row,
            bay: bay,
            tier: tier,
            lengthFt: lengthFt,
            spanCells: span,
            cellIds: cellIds,
            x: pos.x,
            y: pos.y,
            rotation: pos.rotation,
            labelRotationDeg: numberOr(raw.labelRotationDeg, anchor.labelRotationDeg) === 180 ? 180 : 0,
            z: 0
        };
    }

    function markOccupancy(pl) {
        var i;
        for (i = 0; i < pl.cellIds.length; i++) {
            occupancy[occupancyKey(pl.cellIds[i], pl.tier)] = pl.id;
        }
    }

    function topPlacementForCell(cellId) {
        var cell = cellById[String(cellId)];
        if (!cell || cell.count <= 0) {
            return null;
        }
        var t;
        for (t = cell.count; t >= 1; t--) {
            var pid = occupancy[occupancyKey(cell.id, t)];
            if (pid && placementById[pid]) {
                return placementById[pid];
            }
        }
        return null;
    }

    var sourcePlacements = sceneData.placements instanceof Array ? sceneData.placements : [];
    for (var spi = 0; spi < sourcePlacements.length; spi++) {
        var sourcePlacement = normalizePlacement(sourcePlacements[spi]);
        if (placementById[sourcePlacement.id]) {
            throw new Error('Повтор размещения "' + sourcePlacement.id + '".');
        }
        placements.push(sourcePlacement);
        placementById[sourcePlacement.id] = sourcePlacement;
        markOccupancy(sourcePlacement);
    }
    recomputeCellCounts();

    function normalizeSceneObject(sourceObject) {
        var so = sourceObject || {};
        var oid = String(so.id == null ? '' : so.id);
        var kind = String(so.kind == null ? '' : so.kind);
        if (!oid || !kind || kind === 'cell' || kind === 'placement') {
            return null;
        }
        return {
            id: oid,
            kind: kind,
            name: String(so.name || oid),
            x: numberOr(so.x, 0),
            y: numberOr(so.y, 0),
            z: numberOr(so.z, 0),
            rotation: numberOr(so.rotation, 0),
            color: String(so.color || '#94a3b8'),
            size: so.size || {},
            load: String(so.load || 'none'),
            length: numberOr(so.length, numberOr(so.size && so.size.length, 0)),
            vesselId: String(so.vesselId || ''),
            vesselType: String(so.vesselType || ''),
            berthId: String(so.berthId || ''),
            mooringSide: String(so.mooringSide || ''),
            refueling: !!so.refueling,
            locationType: String(so.locationType || ''),
            plannedBerthingAt: String(so.plannedBerthingAt || ''),
            plannedBerthId: String(so.plannedBerthId || ''),
            plannedLoad: plannedContainerCounts(so.plannedLoad),
            plannedUnload: plannedContainerCounts(so.plannedUnload),
            queuePosition: numberOr(so.queuePosition, 0),
            model: String(so.model || ''),
            scale: Math.max(0.01, numberOr(so.scale, 1)),
            azimuthDeg: numberOr(so.azimuthDeg, 0),
            siteId: String(so.siteId || ''),
            movementAxis: String(so.movementAxis || ''),
            positionM: numberOr(so.positionM, 0),
            railInsetM: numberOr(so.railInsetM, 0),
            leftWingM: numberOr(so.leftWingM, 0),
            rightWingM: numberOr(so.rightWingM, 0),
            rails: so.rails instanceof Array ? so.rails : [],
            wagonType: String(so.wagonType || ''),
            equipmentType: String(so.equipmentType || ''),
            withContainer: !!so.withContainer,
            information: String(so.information || ''),
            trainId: String(so.trainId || ''),
            branchId: String(so.branchId || ''),
            loaded: !!so.loaded
        };
    }

    var objects = [];
    var objectById = {};
    var sourceObjects = sceneData.objects instanceof Array ? sceneData.objects : [];
    for (var oi = 0; oi < sourceObjects.length; oi++) {
        var obj = normalizeSceneObject(sourceObjects[oi]);
        if (!obj || objectById[obj.id]) {
            continue;
        }
        objects.push(obj);
        objectById[obj.id] = obj;
    }

    var dpr = 1;
    var view = { scale: 1, offsetX: 40, offsetY: 40 };
    var selected = null;
    var needsDraw = true;
    var panning = false;
    var panLast = null;
    var pointerDown = null;
    var DRAG_THRESHOLD = 4;
    var buildMs = 0;

    var STORAGE_TIER_COLORS = [
        '#514343',
        '#f0a090',
        '#e87868',
        '#d45040',
        '#b83228',
        '#8f2018',
        '#6e1410',
        '#4a0c08'
    ];

    var DISCHARGE_TIER_COLORS = [
        '#3f4954',
        '#9bc8ec',
        '#70addd',
        '#4b91cb',
        '#3175ad',
        '#235b8b',
        '#19466c',
        '#10324f'
    ];

    function colorForCount(count, zoneKind) {
        var palette = zoneKind === 'discharge'
            ? DISCHARGE_TIER_COLORS : STORAGE_TIER_COLORS;
        if (count <= 0) {
            return palette[0];
        }
        var idx = Math.min(palette.length - 1, count);
        return palette[idx];
    }

    function worldToScreen(x, y) {
        return {
            x: view.offsetX + (x - site.minX) * view.scale,
            // Геодезический Y (Northing) растёт к северу, экранный Y — вниз.
            y: view.offsetY + (site.maxY - y) * view.scale
        };
    }

    function screenToWorld(sx, sy) {
        return {
            x: site.minX + (sx - view.offsetX) / view.scale,
            y: site.maxY - (sy - view.offsetY) / view.scale
        };
    }

    function resize() {
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        if (width < 1 || height < 1) {
            return;
        }
        dpr = (window.devicePixelRatio > 1.5) ? 1.5 : (window.devicePixelRatio || 1);
        if (dpr > 2) {
            dpr = 2;
        }
        canvas.width = Math.round(width * dpr);
        canvas.height = Math.round(height * dpr);
        canvas.style.width = width + 'px';
        canvas.style.height = height + 'px';
        needsDraw = true;
    }

    function resetView() {
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        var pad = 40;
        var sx = (width - pad * 2) / site.width;
        var sy = (height - pad * 2) / site.length;
        view.scale = Math.max(0.05, Math.min(sx, sy));
        view.offsetX = (width - site.width * view.scale) / 2;
        view.offsetY = (height - site.length * view.scale) / 2;
        needsDraw = true;
    }

    function ringCentroid(ring) {
        var i;
        var sx = 0;
        var sy = 0;
        var n = ring ? ring.length : 0;
        for (i = 0; i < n; i++) {
            sx += ring[i].x;
            sy += ring[i].y;
        }
        return {
            x: sx / Math.max(1, n),
            y: sy / Math.max(1, n)
        };
    }

    function shorelineCentroid() {
        if (site.waterRings && site.waterRings.length > 0) {
            return ringCentroid(site.waterRings[0]);
        }
        var shore = site.shoreline;
        var i;
        var sx = 0;
        var sy = 0;
        for (i = 0; i < shore.length; i++) {
            sx += shore[i].x;
            sy += shore[i].y * 0.35;
        }
        return {
            x: sx / Math.max(1, shore.length),
            y: Math.max(24, sy / Math.max(1, shore.length))
        };
    }

    function beginWaterPolygons() {
        var rings = site.waterRings;
        var ri;
        var i;
        var p;
        if (!(rings && rings.length > 0)) {
            return false;
        }
        ctx.beginPath();
        for (ri = 0; ri < rings.length; ri++) {
            var ring = rings[ri];
            for (i = 0; i < ring.length; i++) {
                p = worldToScreen(ring[i].x, ring[i].y);
                if (i === 0) {
                    ctx.moveTo(p.x, p.y);
                } else {
                    ctx.lineTo(p.x, p.y);
                }
            }
            ctx.closePath();
        }
        return true;
    }

    // Fallback для stub без site.water: замыкание shoreline с нижней кромкой.
    function beginSeaPolygonFallback() {
        var shore = site.shoreline;
        var i;
        var p;
        ctx.beginPath();
        p = worldToScreen(0, 0);
        ctx.moveTo(p.x, p.y);
        p = worldToScreen(site.width, 0);
        ctx.lineTo(p.x, p.y);
        for (i = shore.length - 1; i >= 0; i--) {
            p = worldToScreen(shore[i].x, shore[i].y);
            ctx.lineTo(p.x, p.y);
        }
        ctx.closePath();
    }

    function beginShorePolyline() {
        var shore = site.shoreline;
        var i;
        var p;
        ctx.beginPath();
        for (i = 0; i < shore.length; i++) {
            p = worldToScreen(shore[i].x, shore[i].y);
            if (i === 0) {
                ctx.moveTo(p.x, p.y);
            } else {
                ctx.lineTo(p.x, p.y);
            }
        }
    }

    function drawRoundedSite() {
        var p0 = worldToScreen(site.minX, site.maxY);
        var sw = site.width * view.scale;
        var sh = site.length * view.scale;
        var shore = site.shoreline;
        var quayW = site.quayWidth;
        var hasWater = site.waterRings && site.waterRings.length > 0;
        var i;

        // суша
        ctx.fillStyle = '#2a3140';
        ctx.fillRect(p0.x, p0.y, sw, sh);

        // море: явный полигон water, иначе legacy-эвристика по shoreline
        if (hasWater) {
            beginWaterPolygons();
        } else {
            beginSeaPolygonFallback();
        }
        ctx.fillStyle = '#4eb8ff';
        ctx.fill();

        ctx.save();
        if (hasWater) {
            beginWaterPolygons();
        } else {
            beginSeaPolygonFallback();
        }
        ctx.clip();
        ctx.strokeStyle = 'rgba(180, 230, 255, 0.22)';
        ctx.lineWidth = Math.max(1, view.scale * 0.35);
        var wy;
        var step = Math.max(6, 7 * view.scale);
        var maxSeaY = 0;
        var minSeaY = site.length;
        var rippleSource = hasWater ? site.waterRings[0] : shore;
        for (i = 0; i < rippleSource.length; i++) {
            if (rippleSource[i].y > maxSeaY) {
                maxSeaY = rippleSource[i].y;
            }
            if (rippleSource[i].y < minSeaY) {
                minSeaY = rippleSource[i].y;
            }
        }
        var yTopScreen = worldToScreen(0, maxSeaY).y;
        var yBotScreen = worldToScreen(0, minSeaY).y;
        var rippleMin = Math.min(yTopScreen, yBotScreen);
        var rippleMax = Math.max(yTopScreen, yBotScreen);
        for (wy = rippleMin + step * 0.6; wy < rippleMax - 2; wy += step) {
            ctx.beginPath();
            var wx;
            for (wx = 0; wx <= site.width; wx += 8) {
                var sx = p0.x + wx * view.scale;
                var amp = Math.sin((wx + wy) * 0.08) * 1.6 * view.scale;
                if (wx === 0) {
                    ctx.moveTo(sx, wy + amp);
                } else {
                    ctx.lineTo(sx, wy + amp);
                }
            }
            ctx.stroke();
        }
        ctx.restore();

        // причал вдоль shoreline (обводка, не заливка)
        if (quayW > 0 && shore.length >= 2) {
            beginShorePolyline();
            ctx.strokeStyle = '#5c6578';
            ctx.lineWidth = Math.max(3, quayW * view.scale);
            ctx.lineJoin = 'round';
            ctx.lineCap = 'round';
            ctx.stroke();
            beginShorePolyline();
            ctx.strokeStyle = '#9aa3b5';
            ctx.lineWidth = Math.max(1, view.scale * 0.35);
            ctx.stroke();
            ctx.fillStyle = '#3d4658';
            for (i = 0; i < shore.length; i++) {
                var bp = worldToScreen(shore[i].x, shore[i].y);
                ctx.fillRect(
                    bp.x - Math.max(1.5, view.scale),
                    bp.y - Math.max(1.5, view.scale),
                    Math.max(3, 2.2 * view.scale),
                    Math.max(3, 2.4 * view.scale));
            }
        }

        if (view.scale > 0.7) {
            var c = shorelineCentroid();
            var cp = worldToScreen(c.x, c.y);
            ctx.fillStyle = 'rgba(220, 240, 255, 0.55)';
            ctx.font = Math.max(11, Math.min(16, 5.5 * view.scale)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('МОРЕ', cp.x, cp.y);
        }

        ctx.strokeStyle = '#465168';
        ctx.lineWidth = 1;
        ctx.strokeRect(p0.x, p0.y, sw, sh);
    }

    function drawCell(cell) {
        var w = numberOr(cell.length, 6.1);
        var h = numberOr(cell.width, 2.5);
        var ang = cell.rotation * Math.PI / 180;
        var p = worldToScreen(cell.x, cell.y);
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(-ang);
        var sw = w * view.scale;
        var sh = h * view.scale;
        // пустая сетка 20 ft; занятость рисуется placements
        ctx.fillStyle = '#454b55';
        ctx.fillRect(-sw / 2, -sh / 2, sw, sh);
        if (view.scale > 2.5) {
            ctx.strokeStyle = 'rgba(0,0,0,0.2)';
            ctx.lineWidth = 1;
            ctx.strokeRect(-sw / 2, -sh / 2, sw, sh);
        }
        ctx.restore();
    }

    function drawPlacement(pl) {
        var anchorCell = cellById[pl.cellIds[0]];
        var baySize = numberOr(anchorCell.length, 6.1);
        var rowSize = numberOr(anchorCell.width, 2.5);
        var w = baySize * pl.spanCells;
        var h = rowSize;
        var ang = pl.rotation * Math.PI / 180;
        var p = worldToScreen(pl.x, pl.y);
        var stackH = 0;
        var i;
        for (i = 0; i < pl.cellIds.length; i++) {
            var cell = cellById[pl.cellIds[i]];
            if (cell && cell.count > stackH) {
                stackH = cell.count;
            }
        }
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(-ang);
        var sw = w * view.scale;
        var sh = h * view.scale;
        ctx.fillStyle = colorForCount(stackH, pl.zoneKind || anchorCell.zoneKind);
        ctx.fillRect(-sw / 2, -sh / 2, sw, sh);
        ctx.strokeStyle = pl.lengthFt === 40 ? '#1f0a08' : 'rgba(0,0,0,0.35)';
        ctx.lineWidth = pl.lengthFt === 40 ? Math.max(1.5, view.scale * 0.2) : 1;
        ctx.strokeRect(-sw / 2, -sh / 2, sw, sh);
        if (selected && (selected.id === pl.id ||
                (selected.kind === 'cell' && pl.cellIds.indexOf(selected.id) >= 0))) {
            ctx.strokeStyle = '#ffcc33';
            ctx.lineWidth = 2;
            ctx.strokeRect(-sw / 2 - 1, -sh / 2 - 1, sw + 2, sh + 2);
        }
        if (view.scale > 3) {
            ctx.fillStyle = '#fff';
            ctx.font = Math.max(8, Math.min(12, 3.2 * view.scale)) + 'px Segoe UI, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.save();
            ctx.rotate(numberOr(pl.labelRotationDeg, 0) * Math.PI / 180);
            ctx.fillText(String(stackH), 0, 0);
            ctx.restore();
        }
        ctx.restore();
    }

    function drawPolygonOutline(item, fillStyle, strokeStyle) {
        var polygon = item && item.polygon instanceof Array ? item.polygon : [];
        if (polygon.length < 3) {
            return;
        }
        ctx.beginPath();
        for (var i = 0; i < polygon.length; i++) {
            var p = worldToScreen(numberOr(polygon[i].x, 0), numberOr(polygon[i].y, 0));
            if (i === 0) {
                ctx.moveTo(p.x, p.y);
            } else {
                ctx.lineTo(p.x, p.y);
            }
        }
        ctx.closePath();
        if (fillStyle) {
            ctx.fillStyle = fillStyle;
            ctx.fill();
        }
        ctx.strokeStyle = strokeStyle || '#8192aa';
        ctx.lineWidth = Math.max(1, view.scale * 0.12);
        ctx.stroke();
    }

    function buildingTextColor(color) {
        var match = /^#([0-9a-f]{6})$/i.exec(String(color || ''));
        if (!match) {
            return '#ffffff';
        }
        var value = parseInt(match[1], 16);
        var red = (value >> 16) & 255;
        var green = (value >> 8) & 255;
        var blue = value & 255;
        return (red * 299 + green * 587 + blue * 114) / 1000 >= 150
            ? '#111827' : '#ffffff';
    }

    function drawBuilding(building) {
        var outline = (selected && selected.id === building.id) ? '#fbbf24' : '#1f2937';
        drawPolygonOutline(building, building.color, outline);
        var center = worldToScreen(building.x, building.y);
        ctx.save();
        ctx.font = 'bold ' + Math.max(10, Math.min(16, 4 * view.scale)) + 'px Segoe UI, sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.lineJoin = 'round';
        ctx.strokeStyle = buildingTextColor(building.color) === '#ffffff'
            ? 'rgba(0,0,0,0.72)' : 'rgba(255,255,255,0.72)';
        ctx.lineWidth = 3;
        ctx.strokeText(building.name, center.x, center.y);
        ctx.fillStyle = buildingTextColor(building.color);
        ctx.fillText(building.name, center.x, center.y);
        ctx.restore();
    }

    function drawTopologyRect(item, fillStyle, strokeStyle) {

        ctx.restore();
    }

    function scoutHasContainer(obj) {

    function scoutVehicleLayout(obj) {
        var type = obj && obj.equipmentType != null ? String(obj.equipmentType) : '';
        var layout = SCOUT_VEHICLE_LAYOUTS[type] || { key: 'dot', hw: 6, hh: 6 };
        if (scoutHasContainer(obj) && layout.loadedHw) {
            return {
                key: layout.key,
                hw: layout.loadedHw,
                hh: layout.loadedHh,
                withContainer: true
            };

        return {
            key: layout.key,
            hw: layout.hw,
            hh: layout.hh,
            withContainer: false
        };
    }

    function equipmentBounds(obj) {
        var kind = obj.kind || '';
        var size = obj.size || {};
        if (kind === 'warehouse') {
            return {
                hw: numberOr(size.length, 30) * 0.5,
                hh: numberOr(size.width, 18) * 0.5
            };
        }
        if (kind === 'vessel') {
            var vesselHh = vesselBeam(obj) * 0.55;
            if (obj.refueling) {
                vesselHh = vesselBeam(obj) * 0.95;
            }
            return {
                hw: vesselLength(obj) * 0.52,
                hh: vesselHh
            };
        }
        if (kind === 'cranePortal') {
            var portalScale = Math.max(0.01, numberOr(obj.scale, 1));
            return {
                hw: numberOr(size.length, 7) * 0.7 * portalScale,
                hh: (numberOr(size.outreach, 21) + numberOr(size.gauge, 9) * 0.5) * 0.55 * portalScale
            };
        }
        if (kind === 'craneRtg' || kind === 'craneRmg') {
            var gantrySpan = numberOr(size.span, 32);
            var leftWing = kind === 'craneRmg' ? Math.max(0, numberOr(obj.leftWingM, 0)) : 0;
            var rightWing = kind === 'craneRmg' ? Math.max(0, numberOr(obj.rightWingM, 0)) : 0;
            return {
                // После поворота +90°: span поперёк площадки (X), length вдоль хода (Y).
                hw: (gantrySpan + leftWing + rightWing) * 0.55,
                hh: numberOr(size.length, 22) * 0.55,
                offsetX: (leftWing - rightWing) * 0.5
            };
        }
        if (kind === 'craneReachStacker') {
            return { hw: 14, hh: 5 };
        }
        if (kind === 'craneMobile') {
            return { hw: 12, hh: 6 };
        }
        if (kind === 'railway') {
            return { hw: numberOr(size.length, 100) * 0.5, hh: 4 };
        }
        if (kind === 'containerTruck') {
            return { hw: 11, hh: 3.2 };
        }
        if (kind === 'dieselLocomotive') {
            return { hw: 9, hh: 2.2 };
        }
        if (kind === 'railWagon') {
            return {
                hw: numberOr(size.length, 14) * 0.55,
                hh: numberOr(size.width, 3.1) * 0.7
            };
        }
        if (kind.indexOf('wagon') === 0) {
            return {
                hw: numberOr(size.length, 14) * 0.55,
                hh: numberOr(size.width, 3.1) * 0.7
            };
        }
        if (kind === 'scoutVehicle') {
            var scoutLayout = scoutVehicleLayout(obj);
            return { hw: scoutLayout.hw, hh: scoutLayout.hh };
        }
        return { hw: 6, hh: 6 };
    }

    function drawVesselCargoContainer(x0, y0, len, beam) {
        var cols = 4;
        var rows = 2;
        var gapX = len * 0.015;
        var gapY = beam * 0.04;
        var cw = (len * 0.54 - gapX * (cols - 1)) / cols;
        var ch = (beam * 0.44 - gapY * (rows - 1)) / rows;
        var colors = ['#2563eb', '#1d4ed8', '#0ea5e9', '#0369a1'];
        var c;
        var r;
        for (r = 0; r < rows; r++) {
            for (c = 0; c < cols; c++) {
                ctx.fillStyle = colors[(c + r) % colors.length];
                strokeFillRect(
                    x0 + len * 0.16 + c * (cw + gapX),
                    y0 + beam * 0.28 + r * (ch + gapY),
                    cw,
                    ch);
            }
        }
    }

    function drawGantryIcon(obj, s, onRails) {
        // После rotation+90: локальный X — поперёк площадки (пролёт), Y — вдоль хода.
        var span = numberOr(obj.size.span, 32) * s;
        var len = numberOr(obj.size.length, 22) * s;
        var leg = Math.max(2.5, 1.6 * s);
        var color = obj.color || '#ffd000';
        var left = -span / 2;
        var right = span / 2;
        var leftWing = onRails ? Math.max(0, numberOr(obj.leftWingM, 0)) * s : 0;
        var rightWing = onRails ? Math.max(0, numberOr(obj.rightWingM, 0)) * s : 0;
        // RMG: центр опоры совпадает с рельсом. RTG: опоры целиком внутри пролёта.
        var leftLeg = onRails ? left - leg / 2 : left;
        var rightLeg = onRails ? right - leg / 2 : right - leg;
        var leftInner = leftLeg + leg;
        var rightInner = rightLeg;
        ctx.strokeStyle = '#111';
        ctx.lineWidth = 1;

        // Опорные балки слева/справа — за пределами площадки.
        ctx.fillStyle = '#374151';
        strokeFillRect(leftLeg, -len / 2, leg, len);
        strokeFillRect(rightLeg, -len / 2, leg, len);

        if (onRails) {
            // Узкие ЖД-катки точно на рельсах.
            var bogieW = Math.max(1.2, 0.7 * s);
            var bogieH = Math.max(2.2, 1.3 * s);
            ctx.fillStyle = '#111827';
            strokeFillRect(left - bogieW / 2, -len / 2, bogieW, bogieH);
            strokeFillRect(left - bogieW / 2, len / 2 - bogieH, bogieW, bogieH);
            strokeFillRect(right - bogieW / 2, -len / 2, bogieW, bogieH);
            strokeFillRect(right - bogieW / 2, len / 2 - bogieH, bogieW, bogieH);
        } else {
            // Широкие резиновые колёса снаружи опор.
            var wheelR = Math.max(2.2, 1.35 * s);
            var wheelX = [left - wheelR * 0.15, right + wheelR * 0.15];
            var wheelY = [-len / 2 + wheelR * 1.1, 0, len / 2 - wheelR * 1.1];
            var wxi;
            var wyi;
            ctx.fillStyle = '#111';
            ctx.strokeStyle = '#0b1220';
            ctx.lineWidth = Math.max(1, s * 0.12);
            for (wxi = 0; wxi < wheelX.length; wxi++) {
                for (wyi = 0; wyi < wheelY.length; wyi++) {
                    ctx.beginPath();
                    ctx.arc(wheelX[wxi], wheelY[wyi], wheelR, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.stroke();
                    ctx.beginPath();
                    ctx.arc(wheelX[wxi], wheelY[wyi], wheelR * 0.45, 0, Math.PI * 2);
                    ctx.stroke();
                }
            }
            ctx.strokeStyle = '#111';
            ctx.lineWidth = 1;
        }

        // Вертикальные стойки.
        ctx.fillStyle = color;
        strokeFillRect(leftLeg, -len / 2, leg, len);
        strokeFillRect(rightLeg, -len / 2, leg, len);
        var beam = Math.max(2, 1.4 * s);
        var beamX = left - rightWing;
        var beamW = span + leftWing + rightWing;

        if (onRails) {
            // Две поперечные балки с крыльями и продольная между ними.
            strokeFillRect(beamX, -len / 2, beamW, beam);
            strokeFillRect(beamX, len / 2 - beam, beamW, beam);
            strokeFillRect(-beam / 2, -len / 2 + beam, beam, len - beam * 2);
        } else {
            strokeFillRect(left, -beam / 2, span, beam);
            ctx.strokeStyle = '#92400e';
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.moveTo(leftInner, -len / 2 + leg);
            ctx.lineTo(rightInner, len / 2 - leg);
            ctx.moveTo(leftInner, len / 2 - leg);
            ctx.lineTo(rightInner, -len / 2 + leg);
            ctx.stroke();
        }

        // Тележка + спредер.
        ctx.fillStyle = '#1f2937';
        strokeFillRect(-beam * 1.6, -2.2 * s, beam * 3.2, 4.4 * s);
        ctx.strokeStyle = color;
        ctx.lineWidth = Math.max(1.5, s * 0.25);
        ctx.strokeRect(beam * 1.2, -3.5 * s, Math.max(3, 2.2 * s), 7 * s);
    }

    function drawRailwayTrack(track) {
        var points = track.points || [];
        if (points.length < 2) {
            return;
        }
        var gaugePx = Math.max(0.7, track.gaugeM * view.scale * 0.5);
        var sleeperHalfPx = Math.max(1.8, track.gaugeM * view.scale * 0.9);
        var sleeperStepM = 2.2;
        var distanceToSleeperM = 0;
        var i;
        ctx.save();
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
        for (i = 1; i < points.length; i++) {
            var first = worldToScreen(points[i - 1].x, points[i - 1].y);
            var second = worldToScreen(points[i].x, points[i].y);
            var dx = second.x - first.x;
            var dy = second.y - first.y;
            var segmentPx = Math.sqrt(dx * dx + dy * dy);
            var segmentM = Math.sqrt(
                Math.pow(points[i].x - points[i - 1].x, 2) +
                Math.pow(points[i].y - points[i - 1].y, 2));
            if (segmentPx < 0.01 || segmentM < 0.01) {
                continue;
            }
            var ux = dx / segmentPx;
            var uy = dy / segmentPx;
            var nx = -uy;
            var ny = ux;

            if (view.scale >= 0.2) {
                ctx.strokeStyle = '#57534e';
                ctx.lineWidth = Math.max(0.7, 0.18 * view.scale);
                var sleeperM = distanceToSleeperM;
                while (sleeperM <= segmentM) {
                    var sleeperRatio = sleeperM / segmentM;
                    var sleeperX = first.x + dx * sleeperRatio;
                    var sleeperY = first.y + dy * sleeperRatio;
                    ctx.beginPath();
                    ctx.moveTo(sleeperX - nx * sleeperHalfPx, sleeperY - ny * sleeperHalfPx);
                    ctx.lineTo(sleeperX + nx * sleeperHalfPx, sleeperY + ny * sleeperHalfPx);
                    ctx.stroke();
                    sleeperM += sleeperStepM;
                }
                distanceToSleeperM = sleeperM - segmentM;
            }

            ctx.strokeStyle = track.color || '#6b7280';
            ctx.lineWidth = Math.max(1, 0.22 * view.scale);
            ctx.beginPath();
            ctx.moveTo(first.x + nx * gaugePx, first.y + ny * gaugePx);
            ctx.lineTo(second.x + nx * gaugePx, second.y + ny * gaugePx);
            ctx.moveTo(first.x - nx * gaugePx, first.y - ny * gaugePx);
            ctx.lineTo(second.x - nx * gaugePx, second.y - ny * gaugePx);
            ctx.stroke();
        }
        ctx.restore();
    }

    function applyDprTransform() {
        try {
            if (typeof ctx.setTransform === 'function') {
                ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

        } catch (ignoreSet) {}
        try {

            }
        } catch (ignoreScale) {}
    }

        }
        for (i = 0; i < objects.length; i++) {

            }
        }
        needsDraw = false;
    }

    function pointInPolygon(world, polygon) {
        if (!(polygon instanceof Array) || polygon.length < 3) {
            return false;
        }
        var inside = false;
        var x = world.x;
        var y = world.y;
        var j = polygon.length - 1;
        for (var i = 0; i < polygon.length; i++) {
            var xi = numberOr(polygon[i].x, 0);
            var yi = numberOr(polygon[i].y, 0);
            var xj = numberOr(polygon[j].x, 0);
            var yj = numberOr(polygon[j].y, 0);
            var intersects = ((yi > y) !== (yj > y)) &&
                (x < (xj - xi) * (y - yi) / ((yj - yi) || 1e-12) + xi);
            if (intersects) {
                inside = !inside;
            }
            j = i;
        }
        return inside;
    }

    function hitTestBuildingFromWorld(world) {
        for (var bi = buildings.length - 1; bi >= 0; bi--) {
            if (pointInPolygon(world, buildings[bi].polygon)) {
                return buildings[bi];
            }
        }
        return null;
    }

    function hitTestCellFromWorld(world) {
        for (var ci = cells.length - 1; ci >= 0; ci--) {
            var cell = cells[ci];
            var ang = numberOr(cell.rotation, 0) * Math.PI / 180;
            var cosA = Math.cos(ang);
            var sinA = Math.sin(ang);
            var dx = world.x - numberOr(cell.x, 0);
            var dy = world.y - numberOr(cell.y, 0);
            var lx = dx * cosA + dy * sinA;
            var ly = -dx * sinA + dy * cosA;
            if (Math.abs(lx) <= numberOr(cell.length, 6.1) * 0.5 &&
                    Math.abs(ly) <= numberOr(cell.width, 2.5) * 0.5) {
                return cell;
            }
        }
        return null;
    }

    function hitTest(sx, sy) {
        var world = screenToWorld(sx, sy);
        var i;
        var pass;
        for (pass = 0; pass < 2; pass++) {
            var cranesOnly = pass === 0;
            for (i = objects.length - 1; i >= 0; i--) {
                var obj = objects[i];
                if (isCraneKind(obj.kind) !== cranesOnly) {
                    continue;
                }
                var b = equipmentBounds(obj);
                var ang = (obj.rotation || 0) * Math.PI / 180;
                var cosA = Math.cos(ang);
                var sinA = Math.sin(ang);
                var lx = (world.x - obj.x) * cosA + (world.y - obj.y) * sinA;
                var ly = -(world.x - obj.x) * sinA + (world.y - obj.y) * cosA;
                var pad = (obj.kind === 'craneReachStacker' || obj.kind === 'craneMobile') ? 2 : 0.5;
                if (Math.abs(lx - numberOr(b.offsetX, 0)) <= b.hw + pad && Math.abs(ly) <= b.hh + pad) {
                    return obj;
                }
            }
        }

        var bestPl = null;
        var bestPlDist = 1e9;
        for (i = 0; i < topPlacementList.length; i++) {
            var top = topPlacementList[i];
            var topCell = cellById[top.cellIds[0]];
            var baySize = numberOr(topCell.length, 6.1);
            var rowSize = numberOr(topCell.width, 2.5);
            var hw = baySize * top.spanCells * 0.5;
            var hh = rowSize * 0.5;
            var pAng = top.rotation * Math.PI / 180;
            var pCos = Math.cos(pAng);
            var pSin = Math.sin(pAng);
            var plx = (world.x - top.x) * pCos + (world.y - top.y) * pSin;
            var ply = -(world.x - top.x) * pSin + (world.y - top.y) * pCos;
            if (Math.abs(plx) <= hw && Math.abs(ply) <= hh) {
                var pd = plx * plx + ply * ply;
                if (pd < bestPlDist) {
                    bestPlDist = pd;
                    bestPl = top;
                }
            }
        }
        if (bestPl) {
            return bestPl;
        }
        var cellHit = hitTestCellFromWorld(world);
        if (cellHit) {
            return cellHit;
        }
        return hitTestBuildingFromWorld(world);
    }

    function notify1C(target) {
        if (!bridge || !target) {
            return;
        }
        bridge.setAttribute('href', '#pick:' + encodeURIComponent(target.id));
        bridge.setAttribute('data-event', 'pick');
        bridge.setAttribute('data-id', target.id);
        bridge.setAttribute('data-kind', target.kind);
        bridge.setAttribute('data-x', JSON.stringify(target.x));
        bridge.setAttribute('data-y', JSON.stringify(target.y));
        bridge.setAttribute('data-z', JSON.stringify(target.z || 0));
        if (target.kind === 'cell') {
            bridge.setAttribute('data-count', JSON.stringify(target.count));
            bridge.setAttribute('data-block', target.blockId);
            bridge.setAttribute('data-row', JSON.stringify(target.row));
            bridge.setAttribute('data-bay', JSON.stringify(target.bay));
        }
        if (target.kind === 'placement') {
            bridge.setAttribute('data-lengthft', JSON.stringify(target.lengthFt));
            bridge.setAttribute('data-tier', JSON.stringify(target.tier));
            bridge.setAttribute('data-block', target.blockId);
            bridge.setAttribute('data-row', JSON.stringify(target.row));
            bridge.setAttribute('data-bay', JSON.stringify(target.bay));
            bridge.setAttribute('data-cells', (target.cellIds || []).join(','));
        }
        if (typeof bridge.click === 'function') {
            bridge.click();
        } else {
            var evt = document.createEvent('MouseEvents');
            evt.initEvent('click', true, true);
            bridge.dispatchEvent(evt);
        }
    }

    function selectTarget(target) {
        selected = target || null;
        needsDraw = true;
        if (target) {
            notify1C(target);
        }
    }

    function focusObject(id) {
        id = String(id == null ? '' : id);
        var target = placementById[id] || cellById[id] || objectById[id] || buildingById[id];
        if (!target && cellById[id]) {
            target = topPlacementForCell(id) || cellById[id];
        }
        if (!target) {
            // поиск ячейки, входящей в placement
            var pi;
            for (pi = 0; pi < placements.length; pi++) {
                if (placements[pi].cellIds && placements[pi].cellIds.indexOf(id) >= 0) {
                    target = placements[pi];
                    break;
                }
            }
        }
        if (!target) {
            throw new Error('Объект с id "' + id + '" не найден.');
        }
        selectTarget(target);
        var width = viewport.clientWidth || 800;
        var height = viewport.clientHeight || 600;
        view.offsetX = width / 2 - (target.x - site.minX) * view.scale;
        view.offsetY = height / 2 - (site.maxY - target.y) * view.scale;
        needsDraw = true;
        return target;
    }

    function updateScoutVehicles(sourceVehicles) {
        if (!(sourceVehicles instanceof Array)) {
            throw new Error('Для обновления техники ожидается массив объектов.');
        }

        var nextObjects = [];
        var nextObjectById = {};
        var objectIndex;
        for (objectIndex = 0; objectIndex < objects.length; objectIndex++) {
            if (objects[objectIndex].kind === 'scoutVehicle') {
                continue;
            }
            nextObjects.push(objects[objectIndex]);
            nextObjectById[objects[objectIndex].id] = objects[objectIndex];
        }

        for (objectIndex = 0; objectIndex < sourceVehicles.length; objectIndex++) {
            var vehicle = normalizeSceneObject(sourceVehicles[objectIndex]);
            if (!vehicle || vehicle.kind !== 'scoutVehicle') {
                throw new Error('Некорректный объект техники в позиции ' + objectIndex + '.');
            }
            if (nextObjectById[vehicle.id]) {
                throw new Error('Повтор идентификатора объекта "' + vehicle.id + '".');
            }
            nextObjects.push(vehicle);
            nextObjectById[vehicle.id] = vehicle;
        }

        var selectedScoutId = selected && selected.kind === 'scoutVehicle' ? selected.id : '';
        objects = nextObjects;
        objectById = nextObjectById;
        if (selectedScoutId) {
            selected = objectById[selectedScoutId] || null;
        }
        needsDraw = true;
        return sourceVehicles.length;
    }

    function updatePlacements(sourcePlacements) {
        if (!(sourcePlacements instanceof Array)) {
            throw new Error('Для обновления контейнеров ожидается массив размещений.');
        }

        var nextPlacements = [];
        var nextPlacementById = {};
        var nextOccupancy = {};
        var placementIndex;
        for (placementIndex = 0; placementIndex < sourcePlacements.length; placementIndex++) {
            var placement = normalizePlacement(sourcePlacements[placementIndex]);
            if (nextPlacementById[placement.id]) {
                throw new Error('Повтор размещения "' + placement.id + '".');
            }
            var cellIndex;
            for (cellIndex = 0; cellIndex < placement.cellIds.length; cellIndex++) {
                var key = occupancyKey(placement.cellIds[cellIndex], placement.tier);
                if (nextOccupancy[key]) {
                    throw new Error('Повторное занятие ячейки "' + key + '".');
                }
                nextOccupancy[key] = placement.id;
            }
            nextPlacements.push(placement);
            nextPlacementById[placement.id] = placement;
        }

        var selectedPlacementId = selected && selected.kind === 'placement' ? selected.id : '';
        placements = nextPlacements;
        placementById = nextPlacementById;
        occupancy = nextOccupancy;
        recomputeCellCounts();
        if (selectedPlacementId) {
            selected = placementById[selectedPlacementId] || null;
        }
        needsDraw = true;
        return sourcePlacements.length;
    }

    function applyCommand(command) {
        command = command || {};
        if (command.op === 'focus') {
            return focusObject(command.id);
        }
        if (command.op === 'updateScoutVehicles') {
            return updateScoutVehicles(command.objects);
        }
        if (command.op === 'updateVessels') {
            return updateVessels(command.objects);
        }
        if (command.op === 'updateRailWagons') {
            return updateRailWagons(command.objects);
        }
        if (command.op === 'updatePlacements') {
            return updatePlacements(command.objects);
        }
        if (command.op === 'setDrawRailwayBranches') {
            drawRailwayBranches = Boolean(command.visible);
            needsDraw = true;
            return drawRailwayBranches;
        }
        if (command.op === 'resize') {
            resize();
            return true;
        }
        throw new Error('Неизвестная операция "' + String(command.op || '') + '".');
    }

    function processBridgeCommand(event) {
        if (event) {
            if (event.preventDefault) {
                event.preventDefault();
            }
            if (event.stopPropagation) {
                event.stopPropagation();
            }
        }
        if (!commandBridge) {
            return false;
        }
        var sequence = commandBridge.getAttribute('data-seq') || '';
        if (!sequence || sequence === processBridgeCommand.lastSequence) {
            return false;
        }
        processBridgeCommand.lastSequence = sequence;
        try {
            applyCommand(JSON.parse(commandBridge.getAttribute('data-payload') || ''));
            commandBridge.setAttribute('data-error', '');
        } catch (commandError) {
            commandBridge.setAttribute('data-error', commandError.message || String(commandError));
        }
        commandBridge.setAttribute('data-ack', sequence);
        return false;
    }
    processBridgeCommand.lastSequence = '';

    window.terminalPlan2d = {
        applyCommand: applyCommand,
        focusObject: focusObject,
        updateScoutVehicles: updateScoutVehicles,
        getCell: function (id) { return cellById[String(id)] || null; },
        getPlacement: function (id) { return placementById[String(id)] || null; },
        getObject: function (id) {
            id = String(id);
            return objectById[id] || buildingById[id] || null;
        }
    };

    if (commandBridge) {
        if (commandBridge.addEventListener) {
            commandBridge.addEventListener('click', processBridgeCommand, false);
        } else if (commandBridge.attachEvent) {
            commandBridge.attachEvent('onclick', processBridgeCommand);
        } else {
            commandBridge.onclick = processBridgeCommand;
        }
        commandBridge.setAttribute('data-ready', 'true');
    }

    function bind(el, eventName, handler, useCapture) {
        if (!el) {
            return;
        }
        if (el.addEventListener) {
            try {
                el.addEventListener(eventName, handler, !!useCapture);
            } catch (ignoreBind) {
                el.addEventListener(eventName, handler);
            }
        } else if (el.attachEvent) {
            el.attachEvent('on' + eventName, handler);
        }
    }

    function onWheel(e) {
        if (!e) { e = window.event; }
        if (e.preventDefault) { e.preventDefault(); }
        var rect = canvas.getBoundingClientRect();
        var sx = e.clientX - rect.left;
        var sy = e.clientY - rect.top;
        var before = screenToWorld(sx, sy);
        var delta = e.deltaY != null ? e.deltaY : (e.wheelDelta ? -e.wheelDelta : 0);
        var factor = delta > 0 ? 0.9 : 1.1;
        view.scale = Math.max(0.05, Math.min(20, view.scale * factor));
        view.offsetX = sx - (before.x - site.minX) * view.scale;
        view.offsetY = sy - (site.maxY - before.y) * view.scale;
        needsDraw = true;
        return false;
    }

    function bootDraw() {
        resize();
        resetView();
        needsDraw = true;
        draw();
    }

    // Сначала рисуем — до API, которые могут упасть в WebKit 1С.
    bootDraw();
    setTimeout(bootDraw, 50);
    setTimeout(bootDraw, 250);

    bind(canvas, 'mousedown', function (e) {
        if (!e) { e = window.event; }
        if (e.button !== 0) { return; }
        panning = true;
        panLast = { x: e.clientX, y: e.clientY };
        pointerDown = { x: e.clientX, y: e.clientY, moved: false };
        viewport.className = (viewport.className || '').replace(/\bpanning\b/g, '') + ' panning';
        if (e.preventDefault) { e.preventDefault(); }
    });

    bind(window, 'mousemove', function (e) {
        if (!e) { e = window.event; }
        if (!panning || !panLast) { return; }
        var dx = e.clientX - panLast.x;
        var dy = e.clientY - panLast.y;
        panLast = { x: e.clientX, y: e.clientY };
        if (pointerDown) {
            var tx = e.clientX - pointerDown.x;
            var ty = e.clientY - pointerDown.y;
            if (tx * tx + ty * ty > DRAG_THRESHOLD * DRAG_THRESHOLD) {
                pointerDown.moved = true;
            }
        }
        view.offsetX += dx;
        view.offsetY += dy;
        needsDraw = true;
    });

    bind(window, 'mouseup', function (e) {
        if (!e) { e = window.event; }
        if (e.button !== 0) { return; }
        var wasClick = pointerDown && !pointerDown.moved;
        var clickX = e.clientX;
        var clickY = e.clientY;
        panning = false;
        panLast = null;
        pointerDown = null;
        viewport.className = (viewport.className || '').replace(/\bpanning\b/g, '').replace(/\s+/g, ' ');
        if (wasClick) {
            var rect = canvas.getBoundingClientRect();
            selectTarget(hitTest(clickX - rect.left, clickY - rect.top));
        }
    });

    bind(canvas, 'mousewheel', onWheel);
    bind(canvas, 'wheel', onWheel);
    canvas.onmousewheel = onWheel;

    bind(document.getElementById('btn-reset'), 'click', function () {
        resetView();
        needsDraw = true;
        draw();
    });

    bind(window, 'resize', function () {
        resize();
        needsDraw = true;
    });
    try {
        if (typeof ResizeObserver !== 'undefined') {
            new ResizeObserver(function () {
                resize();
                needsDraw = true;
            }).observe(viewport);
        }
    } catch (ignoreRo) {}

    var raf = window.requestAnimationFrame || window.webkitRequestAnimationFrame ||
        function (cb) { return setTimeout(function () { cb(); }, 33); };

    function loop() {
        if (needsDraw) {
            draw();
        }
        raf(loop);
    }
    raf(loop);
}

function startTerminalPlan2D() {
    try {
        initTerminalPlan2D();
    } catch (bootError) {
        var errNode = document.getElementById('plan-error');
        var msg = 'Ошибка плана: ' + (bootError && bootError.message
            ? bootError.message : String(bootError));
        if (errNode) {
            errNode.style.display = 'block';
            errNode.textContent = msg;
        }
    }
}

if (document.readyState === 'loading') {
    if (document.addEventListener) {
        document.addEventListener('DOMContentLoaded', startTerminalPlan2D, false);
    } else {
        setTimeout(startTerminalPlan2D, 0);
    }
} else {
    startTerminalPlan2D();
}
