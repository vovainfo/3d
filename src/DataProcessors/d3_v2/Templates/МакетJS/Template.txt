<script>
function initWarehouse3D() {
    var viewport = document.getElementById('viewport');
    var statusEl = document.getElementById('status');
    var bridge = document.getElementById('pick-bridge');
    var commandBridge = document.getElementById('scene-command-bridge');
    var errEl = document.getElementById('webgl-error');
    var dataEl = document.getElementById('scene-data');

    function showError(text) {
        if (errEl) {
            errEl.style.display = 'flex';
            errEl.textContent = text;
        }
    }

    if (!viewport || typeof THREE === 'undefined') {
        showError('Three.js не загружен.');
        return;
    }

    var sceneData;
    try {
        var jsonText = dataEl ? (dataEl.textContent || dataEl.innerText || '') : '';
        sceneData = JSON.parse(jsonText);
    } catch (parseError) {
        showError('Ошибка JSON сцены: ' + parseError.message);
        return;
    }

    if (!sceneData || Number(sceneData.version) !== 1 || sceneData.units !== 'm') {
        showError('Неподдерживаемая версия или единицы сцены.');
        return;
    }

    function numberOr(value, fallback) {
        var n = Number(value);
        return isFinite(n) ? n : fallback;
    }

    function positiveOr(value, fallback) {
        var n = numberOr(value, fallback);
        return n > 0 ? n : fallback;
    }

    var site = sceneData.site || {};
    site.width = positiveOr(site.width, 100);
    site.length = positiveOr(site.length, 100);
    site.gridStep = positiveOr(site.gridStep, 1);

    var sourceObjects = sceneData.objects instanceof Array ? sceneData.objects : [];
    var objects = [];
    var objectById = {};
    var invalidCount = 0;

    function normalizeObject(source) {
        source = source || {};
        var id = String(source.id == null ? '' : source.id);
        var kind = String(source.kind == null ? '' : source.kind);
        var x = Number(source.x);
        var y = Number(source.y);
        var z = Number(source.z);
        if (!id || !kind || !isFinite(x) || !isFinite(y) || !isFinite(z)) {
            return null;
        }
        return {
            id: id,
            kind: kind,
            name: String(source.name || id),
            x: x,
            y: y,
            z: z,
            rotation: numberOr(source.rotation, 0),
            color: String(source.color || '#607d8b'),
            size: source.size || {},
            load: String(source.load || 'none'),
            loadColor: String(source.loadColor || '#1769aa'),
            blockId: source.blockId == null ? '' : String(source.blockId),
            row: source.row,
            bay: source.bay,
            tier: source.tier,
            status: source.status == null ? '' : String(source.status),
            observedAt: source.observedAt == null ? '' : String(source.observedAt),
            source: source,
            bounds: new THREE.Box3(),
            instanceLayer: null,
            instanceIndex: -1
        };
    }

    for (var oi = 0; oi < sourceObjects.length; oi++) {
        var obj = normalizeObject(sourceObjects[oi]);
        if (!obj || objectById[obj.id]) {
            invalidCount++;
            continue;
        }
        obj.objectIndex = objects.length;
        objectById[obj.id] = obj;
        objects.push(obj);
    }

    var scene = new THREE.Scene();
    scene.background = new THREE.Color(0x1a1d24);

    var camera = new THREE.PerspectiveCamera(60, 1, 0.5, Math.max(1000, site.width + site.length));
    var renderer;
    try {
        renderer = new THREE.WebGLRenderer({
            antialias: true,
            alpha: false,
            failIfMajorPerformanceCaveat: false
        });
    } catch (rendererError) {
        showError('WebGL недоступен в этом клиенте.');
        return;
    }
    renderer.setPixelRatio(1);
    renderer.setClearColor(0x1a1d24, 1);
    renderer.domElement.tabIndex = 0;
    viewport.appendChild(renderer.domElement);

    // На встроенной графике стоимость кадра в первую очередь зависит от площади
    // framebuffer. На экранах выше Full HD рендерим в меньшем внутреннем
    // разрешении, оставляя CSS-размер canvas равным размеру окна.
    var MAX_RENDER_PIXELS = 1920 * 1080;
    var renderScale = 1;

    // --- Площадка в метрах ---
    var ground = new THREE.Mesh(
        new THREE.PlaneGeometry(site.width, site.length),
        new THREE.MeshBasicMaterial({ color: 0x2e3544 })
    );
    ground.rotation.x = -Math.PI / 2;
    ground.position.set(site.width / 2, 0, site.length / 2);
    scene.add(ground);

    function createMeterGrid() {
        var points = [];
        var x;
        var y;
        for (x = 0; x <= site.width + 0.0001; x += site.gridStep) {
            points.push(x, 0.01, 0, x, 0.01, site.length);
        }
        for (y = 0; y <= site.length + 0.0001; y += site.gridStep) {
            points.push(0, 0.01, y, site.width, 0.01, y);
        }
        var geometry = new THREE.BufferGeometry();
        geometry.setAttribute('position', new THREE.Float32BufferAttribute(points, 3));
        return new THREE.LineSegments(
            geometry,
            new THREE.LineBasicMaterial({ color: 0x465168, transparent: true, opacity: 0.55 })
        );
    }
    scene.add(createMeterGrid());

    // --- Примитивы процедурных моделей ---
    var UNIT_BOX = new THREE.BoxGeometry(1, 1, 1);
    var UNIT_CYLINDER = new THREE.CylinderGeometry(1, 1, 1, 8, 1, false);

    function createGableGeometry() {
        var geometry = new THREE.BufferGeometry();
        geometry.setAttribute('position', new THREE.Float32BufferAttribute([
            -0.5, 0, -0.5,  -0.5, 0, 0.5,  -0.5, 1, 0,
             0.5, 0, -0.5,   0.5, 0, 0.5,   0.5, 1, 0
        ], 3));
        geometry.setIndex([
            0, 1, 2,  3, 5, 4,
            0, 3, 4,  0, 4, 1,
            1, 4, 5,  1, 5, 2,
            2, 5, 3,  2, 3, 0
        ]);
        geometry.computeVertexNormals();
        return geometry;
    }
    var UNIT_GABLE = createGableGeometry();

    function colorValue(value, fallback) {
        try {
            return new THREE.Color(value || fallback);
        } catch (ignore) {
            return new THREE.Color(fallback);
        }
    }

    function shadeColor(value, factor) {
        var c = colorValue(value, '#607d8b');
        c.r = Math.min(1, Math.max(0, c.r * factor));
        c.g = Math.min(1, Math.max(0, c.g * factor));
        c.b = Math.min(1, Math.max(0, c.b * factor));
        return '#' + c.getHexString();
    }

    function makePart(geometry, sx, sy, sz, px, py, pz, color, rx, ry, rz) {
        var position = new THREE.Vector3(px || 0, py || 0, pz || 0);
        var rotation = new THREE.Euler(rx || 0, ry || 0, rz || 0, 'XYZ');
        var scale = new THREE.Vector3(sx, sy, sz);
        var quaternion = new THREE.Quaternion().setFromEuler(rotation);
        var matrix = new THREE.Matrix4();
        matrix.compose(position, quaternion, scale);
        return { geometry: geometry, matrix: matrix, color: color };
    }

    function boxPart(length, height, width, x, y, z, color, rx, ry, rz) {
        return makePart(UNIT_BOX, length, height, width, x, y, z, color, rx, ry, rz);
    }

    function cylinderPart(radius, length, x, y, z, color, rx, ry, rz) {
        return makePart(UNIT_CYLINDER, radius, length, radius, x, y, z, color, rx, ry, rz);
    }

    function addContainerParts(parts, length, width, height, baseY, color, offsetX, offsetZ) {
        var x = offsetX || 0;
        var z = offsetZ || 0;
        parts.push(boxPart(length, height, width, x, baseY + height / 2, z, color));
        var door = shadeColor(color, 0.58);
        parts.push(boxPart(0.035, height * 0.86, width / 2 - 0.04,
            x + length / 2 - 0.02, baseY + height * 0.5, z - width / 4, door));
        parts.push(boxPart(0.035, height * 0.86, width / 2 - 0.04,
            x + length / 2 - 0.02, baseY + height * 0.5, z + width / 4, door));
    }

    var objectFactories = {};

    objectFactories.container20 = function (obj) {
        var parts = [];
        addContainerParts(parts, 6.058, 2.438, 2.591, 0, obj.color, 0, 0);
        return parts;
    };

    objectFactories.container40 = function (obj) {
        var parts = [];
        addContainerParts(parts, 12.192, 2.438, 2.591, 0, obj.color, 0, 0);
        return parts;
    };

    objectFactories.warehouse = function (obj) {
        var length = positiveOr(obj.size.length, 30);
        var width = positiveOr(obj.size.width, 18);
        var height = positiveOr(obj.size.height, 8);
        var wallHeight = height * 0.76;
        var roofHeight = height - wallHeight;
        var wallColor = obj.color;
        var roofColor = shadeColor(obj.color, 1.25);
        var doorColor = shadeColor(obj.color, 0.42);
        return [
            boxPart(length, wallHeight, width, 0, wallHeight / 2, 0, wallColor),
            makePart(UNIT_GABLE, length, roofHeight, width,
                0, wallHeight, 0, roofColor),
            boxPart(0.08, wallHeight * 0.68, Math.min(6, width * 0.45),
                length / 2 - 0.05, wallHeight * 0.34, 0, doorColor)
        ];
    };

    objectFactories.craneRtg = function (obj) {
        var span = positiveOr(obj.size.span, 24);
        var length = positiveOr(obj.size.length, 15);
        var height = positiveOr(obj.size.height, 18);
        var color = obj.color;
        var dark = shadeColor(color, 0.55);
        var parts = [];
        var legX = length * 0.42;
        var legZ = span * 0.47;
        // Опоры доходят до верхних поперечных балок без воздушного зазора.
        var legH = height;
        var braceWidth = legX * 2;
        var braceLength = Math.sqrt(braceWidth * braceWidth + legH * legH);
        var braceAngle = Math.atan2(legH, braceWidth);
        var ix;
        var iz;
        for (ix = -1; ix <= 1; ix += 2) {
            for (iz = -1; iz <= 1; iz += 2) {
                parts.push(boxPart(0.95, legH, 0.95, ix * legX, legH / 2, iz * legZ, color));

                    0x222831, Math.PI / 2, 0, 0));
            }
            // Диагональные фермы на обеих боковых рамах.
            parts.push(boxPart(braceLength, 0.42, 0.35,
                0, legH / 2, ix * legZ, dark, 0, 0, braceAngle));
            parts.push(boxPart(braceLength, 0.42, 0.35,
                0, legH / 2, ix * legZ, dark, 0, 0, -braceAngle));
        }
        parts.push(boxPart(1.2, 1.35, span + 2.0, -legX, height, 0, color));
        parts.push(boxPart(1.2, 1.35, span + 2.0, legX, height, 0, color));
        parts.push(boxPart(length, 1.0, 1.15, 0, height + 1.25, 0, dark));
        parts.push(boxPart(2.1, 0.7, 1.8, 0, height + 0.25, 0, '#334155'));
        parts.push(boxPart(2.4, 1.8, 2.0, -2.8, height - 0.45, 0, '#38546a'));
        parts.push(boxPart(0.08, 1.15, 1.72, -1.56, height - 0.35, 0, '#8ec5df'));
        parts.push(cylinderPart(0.35, 0.8, -legX, height + 2.1, 0, '#ff3030'));
        parts.push(cylinderPart(0.35, 0.8, legX, height + 2.1, 0, '#ff3030'));
        var cableLength = height * 0.49;
        var cableY = height * 0.73;
        for (ix = -1; ix <= 1; ix += 2) {
            for (iz = -1; iz <= 1; iz += 2) {
                parts.push(boxPart(0.07, cableLength, 0.07,
                    ix * 2.2, cableY, iz * 0.9, '#20242c'));
            }
        }
        parts.push(boxPart(5.8, 0.25, 2.5, 0, height * 0.48, 0, '#d9c43b'));
        parts.push(boxPart(0.35, 0.45, 3.2, -2.65, height * 0.46, 0, '#b69f24'));
        parts.push(boxPart(0.35, 0.45, 3.2, 2.65, height * 0.46, 0, '#b69f24'));
        // Галерея обслуживания с ограждением.
        parts.push(boxPart(length * 0.78, 0.14, 1.0,
            0, height + 1.85, -span * 0.34, '#69727d'));
        parts.push(boxPart(length * 0.78, 0.1, 0.1,
            0, height + 2.65, -span * 0.72 / 2, '#d7dde3'));
        for (ix = -2; ix <= 2; ix++) {
            parts.push(boxPart(0.09, 0.85, 0.09,
                ix * length * 0.16, height + 2.25, -span * 0.34, '#d7dde3'));
        }
        // Лестница на одной из опор.
        parts.push(boxPart(0.08, height * 0.68, 0.08,
            -legX - 0.52, height * 0.34, legZ + 0.42, '#c8cdd3'));
        parts.push(boxPart(0.08, height * 0.68, 0.08,
            -legX - 0.52, height * 0.34, legZ - 0.42, '#c8cdd3'));
        for (var rungY = 1.0; rungY < height * 0.68; rungY += 1.25) {
            parts.push(boxPart(0.08, 0.08, 0.9,
                -legX - 0.52, rungY, legZ, '#c8cdd3'));
        }
        return parts;
    };

    objectFactories.railway = function (obj) {
        var length = positiveOr(obj.size.length, 100);
        var gauge = positiveOr(obj.size.gauge, 1.52);
        var sleeperStep = positiveOr(obj.size.sleeperStep, 0.82);
        var curveOffset = numberOr(obj.size.curveOffset, 0);
        var segmentCount = Math.max(8, Math.ceil(length / 2));
        var segmentLength = length / segmentCount;
        var parts = [];

        function curveAt(t) {
            var x = -length / 2 + t * length;
            var z = curveOffset * Math.sin(Math.PI * t);
            var slope = curveOffset * Math.PI / length * Math.cos(Math.PI * t);
            var angle = Math.atan(slope);
            return {
                x: x,
                z: z,
                angle: angle,
                normalX: -Math.sin(angle),
                normalZ: Math.cos(angle)
            };
        }

        // Балласт и рельсы собираются из коротких секций по касательной к кривой.
        for (var segment = 0; segment < segmentCount; segment++) {
            var point = curveAt((segment + 0.5) / segmentCount);
            parts.push(boxPart(segmentLength * 1.06, 0.08, 3.2,
                point.x, 0.04, point.z, '#4b5563', 0, -point.angle, 0));

                parts.push(boxPart(segmentLength * 1.06, 0.16, 0.09,
                    point.x + point.normalX * railOffset,
                    0.25,
                    point.z + point.normalZ * railOffset,
                    '#aeb5bd', 0, -point.angle, 0));
            }
        }

        var sleeperCount = Math.floor(length / sleeperStep);
        for (var i = 0; i < sleeperCount; i++) {

    // --- Совместимый с WebKit батчинг BufferGeometry (инфраструктура/техника) ---
    var renderMeshes = [];
    var objectBatchesByIndex = {};
    var MAX_BATCH_VERTICES = 60000;
    var buildStartedAt = (typeof performance !== 'undefined' && performance.now)
        ? performance.now() : Date.now();
    var buildMs = 0;
    var lastDeltaMs = 0;
    var fpsAccum = 0;
    var fpsFrames = 0;
    var fpsValue = 0;

    var CONTAINER_SIZES = {
        container20: { length: 6.058, width: 2.438, height: 2.591 },
        container40: { length: 12.192, width: 2.438, height: 2.591 }
    };

    function isContainerKind(kind) {
        return kind === 'container20' || kind === 'container40';
    }

    // InstancedMesh в WebKit ПолеHTMLДокумента часто не рисует экземпляры.
    // По умолчанию — проверенный батчинг; instancing включается только явно.
    var useInstancedContainers = !!(sceneData.flags && sceneData.flags.useInstancedContainers);
    var containerRenderMode = useInstancedContainers ? 'instanced' : 'batched';

    function GeometryBuilder() {
        this.positions = [];
        this.colors = [];
        this.indices = [];
        this.edgePositions = [];
        this.ranges = [];
        this.objectIndexes = [];
        this.vertexCount = 0;
        this.skipEdges = false;
    }

    GeometryBuilder.prototype.isEmpty = function () {
        return this.vertexCount === 0;
    };

    GeometryBuilder.prototype.addPart = function (part, worldMatrix, objectIndex) {
        var geometry = part.geometry;
        var position = geometry.attributes.position;
        var normal = geometry.attributes.normal;
        var index = geometry.index;
        if (!position || !normal) {
            return false;
        }
        if (this.vertexCount + position.count > MAX_BATCH_VERTICES && !this.isEmpty()) {
            return false;
        }

        var baseVertex = this.vertexCount;
        var vector = new THREE.Vector3();
        var normalVector = new THREE.Vector3();
        var normalMatrix = new THREE.Matrix3().getNormalMatrix(worldMatrix);
        var baseColor = colorValue(part.color, '#607d8b');

        for (var i = 0; i < position.count; i++) {
            vector.fromBufferAttribute(position, i).applyMatrix4(worldMatrix);
            this.positions.push(vector.x, vector.y, vector.z);

            normalVector.fromBufferAttribute(normal, i).applyMatrix3(normalMatrix).normalize();
            var factor = normalVector.y > 0.5 ? 1.28 :
                (normalVector.x > 0.35 ? 0.72 :
                (normalVector.z > 0.35 ? 0.82 : 0.58));
            this.colors.push(
                Math.min(1, baseColor.r * factor),
                Math.min(1, baseColor.g * factor),
                Math.min(1, baseColor.b * factor)
            );
        }

        var faceStart = this.indices.length / 3;
        if (index) {
            for (i = 0; i < index.count; i++) {
                this.indices.push(baseVertex + index.getX(i));
            }
        } else {
            for (i = 0; i < position.count; i++) {
                this.indices.push(baseVertex + i);
            }
        }
        var faceEnd = this.indices.length / 3;
        var previous = this.ranges.length ? this.ranges[this.ranges.length - 1] : null;
        if (previous && previous.objectIndex === objectIndex && previous.endFace === faceStart) {
            previous.endFace = faceEnd;
        } else {
            this.ranges.push({
                startFace: faceStart,
                endFace: faceEnd,
                objectIndex: objectIndex
            });
            if (!this.objectIndexes.length ||
                    this.objectIndexes[this.objectIndexes.length - 1] !== objectIndex) {
                this.objectIndexes.push(objectIndex);
            }
        }

        if (!geometry.boundingBox) {
            geometry.computeBoundingBox();
        }
        var partBounds = geometry.boundingBox.clone().applyMatrix4(worldMatrix);
        objects[objectIndex].bounds.union(partBounds);

        if (!this.skipEdges) {
            var edgeGeometry = geometry.userData.meterEdges;
            if (!edgeGeometry) {
                edgeGeometry = new THREE.EdgesGeometry(geometry, 20);
                geometry.userData.meterEdges = edgeGeometry;
            }
            var edgePosition = edgeGeometry.attributes.position;
            for (i = 0; i < edgePosition.count; i++) {
                vector.fromBufferAttribute(edgePosition, i).applyMatrix4(worldMatrix);
                this.edgePositions.push(vector.x, vector.y, vector.z);
            }
        }

        this.vertexCount += position.count;
        return true;
    };

    GeometryBuilder.prototype.flush = function () {
        if (this.isEmpty()) {
            return null;
        }
        var geometry = new THREE.BufferGeometry();
        geometry.setAttribute('position', new THREE.Float32BufferAttribute(this.positions, 3));
        geometry.setAttribute('color', new THREE.Float32BufferAttribute(this.colors, 3));
        geometry.setIndex(new THREE.Uint16BufferAttribute(this.indices, 1));
        geometry.computeBoundingSphere();

        var mesh = new THREE.Mesh(
            geometry,
            new THREE.MeshBasicMaterial({ vertexColors: true })
        );
        mesh.userData.ranges = this.ranges;
        mesh.userData.objectIndexes = this.objectIndexes;

        if (this.edgePositions.length) {
            var edgeGeometry = new THREE.BufferGeometry();
            edgeGeometry.setAttribute('position', new THREE.Float32BufferAttribute(this.edgePositions, 3));
            var edges = new THREE.LineSegments(
                edgeGeometry,
                new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.48 })
            );
            edges.raycast = function () {};
            mesh.add(edges);
        }
        scene.add(mesh);
        renderMeshes.push(mesh);
        for (var i = 0; i < this.objectIndexes.length; i++) {
            var objectIndex = this.objectIndexes[i];
            if (!objectBatchesByIndex[objectIndex]) {
                objectBatchesByIndex[objectIndex] = [];
            }
            objectBatchesByIndex[objectIndex].push(mesh);
        }
        return mesh;
    };

    function objectWorldMatrix(obj) {
        var position = new THREE.Vector3(obj.x, obj.z, obj.y);
        var rotation = new THREE.Quaternion().setFromEuler(
            new THREE.Euler(0, -obj.rotation * Math.PI / 180, 0)
        );
        return new THREE.Matrix4().compose(position, rotation, new THREE.Vector3(1, 1, 1));
    }

    function disposeBatch(mesh) {
        if (!mesh) {
            return;
        }
        scene.remove(mesh);
        var meshPosition = renderMeshes.indexOf(mesh);
        if (meshPosition >= 0) {
            renderMeshes.splice(meshPosition, 1);
        }
        var indexes = mesh.userData.objectIndexes || [];
        for (var i = 0; i < indexes.length; i++) {
            var batches = objectBatchesByIndex[indexes[i]] || [];
            var batchPosition = batches.indexOf(mesh);
            if (batchPosition >= 0) {
                batches.splice(batchPosition, 1);
            }
            if (!batches.length) {
                delete objectBatchesByIndex[indexes[i]];
            }
        }
        for (i = 0; i < mesh.children.length; i++) {
            if (mesh.children[i].geometry) {
                mesh.children[i].geometry.dispose();
            }
            if (mesh.children[i].material) {
                mesh.children[i].material.dispose();
            }
        }
        mesh.geometry.dispose();
        mesh.material.dispose();
    }

    function buildObjectBatches(indexes, countInvalid, options) {
        options = options || {};
        var skipContainers = !!options.skipContainers;
        var builder = new GeometryBuilder();
        builder.skipEdges = !!options.skipEdges;
        var builtObjects = 0;
        for (var ii = 0; ii < indexes.length; ii++) {
            var objectIndex = indexes[ii];
            var obj = objects[objectIndex];
            if (!obj) {
                continue;
            }
            if (skipContainers && isContainerKind(obj.kind)) {
                continue;
            }
            var factory = objectFactories[obj.kind];
            if (!factory) {
                if (countInvalid) {
                    invalidCount++;
                }
                continue;
            }
            obj.bounds.makeEmpty();
            var parts;
            try {
                parts = factory(obj);
            } catch (factoryError) {
                if (countInvalid) {
                    invalidCount++;
                }
                continue;
            }
            // Для штабеля достаточно одного бокса — без дверей/рёбер.
            if (isContainerKind(obj.kind) && parts.length) {
                parts = [parts[0]];
            }
            var objectMatrix = objectWorldMatrix(obj);
            for (var pi = 0; pi < parts.length; pi++) {
                var partMatrix = new THREE.Matrix4().multiplyMatrices(objectMatrix, parts[pi].matrix);
                if (!builder.addPart(parts[pi], partMatrix, objectIndex)) {
                    builder.flush();
                    builder = new GeometryBuilder();
                    builder.skipEdges = !!options.skipEdges;
                    builder.addPart(parts[pi], partMatrix, objectIndex);
                }
            }
            builtObjects++;
        }
        builder.flush();
        return builtObjects;
    }

    function rebuildBatchesForObject(objectIndex) {
        var oldBatches = (objectBatchesByIndex[objectIndex] || []).slice();
        var rebuildIndexes = {};
        for (var bi = 0; bi < oldBatches.length; bi++) {
            var batchIndexes = oldBatches[bi].userData.objectIndexes || [];
            for (var ji = 0; ji < batchIndexes.length; ji++) {
                rebuildIndexes[batchIndexes[ji]] = true;
            }
        }
        for (bi = 0; bi < oldBatches.length; bi++) {
            disposeBatch(oldBatches[bi]);
        }
        var indexes = [];
        for (var key in rebuildIndexes) {
            if (Object.prototype.hasOwnProperty.call(rebuildIndexes, key) && objects[Number(key)]) {
                indexes.push(Number(key));
            }
        }
        indexes.sort(function (a, b) { return a - b; });
        var rebuildSkipEdges = false;
        for (var ri = 0; ri < indexes.length; ri++) {
            if (objects[indexes[ri]] && isContainerKind(objects[indexes[ri]].kind)) {
                rebuildSkipEdges = true;
                break;
            }
        }
        buildObjectBatches(indexes, false, { skipEdges: rebuildSkipEdges });
    }

    // --- InstancedMesh слой контейнеров ---
    var containerLayers = {};
    var containerPickMeshes = [];
    var _tmpMatrix = new THREE.Matrix4();
    var _tmpPos = new THREE.Vector3();
    var _tmpQuat = new THREE.Quaternion();
    var _tmpScale = new THREE.Vector3();
    var _tmpColor = new THREE.Color();
    var _tmpBox = new THREE.Box3();
    var _tmpSize = new THREE.Vector3();

    function layerKey(kind, blockId) {
        return kind + '|' + (blockId || '_');
    }

    function ensureContainerLayer(kind, blockId, minCapacity) {
        var key = layerKey(kind, blockId);
        var layer = containerLayers[key];
        var capacity = Math.max(64, minCapacity || 64);
        if (layer && layer.capacity >= capacity) {
            return layer;
        }
        if (layer) {
            capacity = Math.max(capacity, layer.capacity * 2);
            scene.remove(layer.mesh);
            var idx = containerPickMeshes.indexOf(layer.mesh);
            if (idx >= 0) {
                containerPickMeshes.splice(idx, 1);
            }
            layer.mesh.geometry.dispose();
            layer.mesh.material.dispose();
        }
        var size = CONTAINER_SIZES[kind];
        var geometry = new THREE.BoxGeometry(1, 1, 1);
        var material = new THREE.MeshBasicMaterial({ color: 0xffffff });
        var mesh = new THREE.InstancedMesh(geometry, material, capacity);
        // Локальная геометрия — unit-box; без отключения culling весь слой
        // отсекается, т.к. bounding sphere остаётся у начала координат.
        mesh.frustumCulled = false;
        if (THREE.DynamicDrawUsage != null) {
            mesh.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
        }
        mesh.count = 0;
        mesh.userData.layerKey = key;
        mesh.userData.kind = kind;
        scene.add(mesh);
        containerPickMeshes.push(mesh);

        var idByIndex = layer ? layer.idByIndex.slice() : [];
        var freeList = layer ? layer.freeList.slice() : [];
        while (idByIndex.length < capacity) {
            idByIndex.push(null);
        }
        // Hide unused slots.
        _tmpScale.set(0, 0, 0);
        _tmpPos.set(0, -10000, 0);
        _tmpQuat.identity();
        _tmpMatrix.compose(_tmpPos, _tmpQuat, _tmpScale);
        var start = layer ? layer.capacity : 0;
        for (var i = start; i < capacity; i++) {
            mesh.setMatrixAt(i, _tmpMatrix);
            if (typeof mesh.setColorAt === 'function') {
                mesh.setColorAt(i, _tmpColor.set('#607d8b'));
            }
        }
        mesh.instanceMatrix.needsUpdate = true;
        if (mesh.instanceColor) {
            mesh.instanceColor.needsUpdate = true;
        }
        if (layer) {
            // Copy previous instance matrices/colors.
            for (i = 0; i < layer.capacity; i++) {
                layer.mesh.getMatrixAt(i, _tmpMatrix);
                mesh.setMatrixAt(i, _tmpMatrix);
                if (layer.mesh.instanceColor && mesh.instanceColor) {
                    layer.mesh.getColorAt(i, _tmpColor);
                    mesh.setColorAt(i, _tmpColor);
                }
            }
            mesh.count = layer.mesh.count;
            mesh.instanceMatrix.needsUpdate = true;
            if (mesh.instanceColor) {
                mesh.instanceColor.needsUpdate = true;
            }
        }

        layer = {
            key: key,
            kind: kind,
            blockId: blockId || '',
            mesh: mesh,
            idByIndex: idByIndex,
            freeList: freeList,
            capacity: capacity,
            size: size
        };
        containerLayers[key] = layer;
        return layer;
    }

    function updateContainerBounds(obj) {
        var size = CONTAINER_SIZES[obj.kind];
        if (!size) {
            return;
        }
        _tmpPos.set(obj.x, obj.z + size.height / 2, obj.y);
        _tmpQuat.setFromEuler(new THREE.Euler(0, -obj.rotation * Math.PI / 180, 0));
        _tmpScale.set(size.length, size.height, size.width);
        _tmpMatrix.compose(_tmpPos, _tmpQuat, _tmpScale);
        obj.bounds.makeEmpty();
        _tmpBox.setFromCenterAndSize(
            _tmpPos,
            _tmpSize.set(size.length, size.height, size.width)
        );
        // Approximate AABB without full OBB for highlight.
        var halfL = size.length * 0.5;
        var halfW = size.width * 0.5;
        var halfH = size.height * 0.5;
        var cosR = Math.cos(-obj.rotation * Math.PI / 180);
        var sinR = Math.sin(-obj.rotation * Math.PI / 180);
        var extX = Math.abs(cosR) * halfL + Math.abs(sinR) * halfW;
        var extZ = Math.abs(sinR) * halfL + Math.abs(cosR) * halfW;
        obj.bounds.min.set(_tmpPos.x - extX, obj.z, _tmpPos.z - extZ);
        obj.bounds.max.set(_tmpPos.x + extX, obj.z + size.height, _tmpPos.z + extZ);
    }

    function writeContainerMatrix(layer, index, obj) {
        var size = layer.size;
        _tmpPos.set(obj.x, obj.z + size.height / 2, obj.y);
        _tmpQuat.setFromEuler(new THREE.Euler(0, -obj.rotation * Math.PI / 180, 0));
        _tmpScale.set(size.length, size.height, size.width);
        _tmpMatrix.compose(_tmpPos, _tmpQuat, _tmpScale);
        layer.mesh.setMatrixAt(index, _tmpMatrix);
        if (typeof layer.mesh.setColorAt === 'function') {
            layer.mesh.setColorAt(index, colorValue(obj.color, '#607d8b'));
        }
        updateContainerBounds(obj);
    }

    function hideContainerSlot(layer, index) {
        _tmpPos.set(0, -10000, 0);
        _tmpQuat.identity();
        _tmpScale.set(0, 0, 0);
        _tmpMatrix.compose(_tmpPos, _tmpQuat, _tmpScale);
        layer.mesh.setMatrixAt(index, _tmpMatrix);
    }

    function allocateContainerSlot(layer) {
        if (layer.freeList.length) {
            return layer.freeList.pop();
        }
        if (layer.mesh.count < layer.capacity) {
            var index = layer.mesh.count;
            layer.mesh.count = index + 1;
            return index;
        }
        var grown = ensureContainerLayer(layer.kind, layer.blockId, layer.capacity * 2);
        return allocateContainerSlot(grown);
    }

    function addContainerInstance(obj) {
        var estimated = 1;
        var layer = ensureContainerLayer(obj.kind, obj.blockId, estimated);
        // Grow eagerly when first filling a large block.
        if (layer.mesh.count + 1 > layer.capacity * 0.9) {
            layer = ensureContainerLayer(obj.kind, obj.blockId, layer.capacity + 1024);
        }
        var index = allocateContainerSlot(layer);
        // allocate may have swapped layer reference after grow
        layer = containerLayers[layerKey(obj.kind, obj.blockId)];
        layer.idByIndex[index] = obj.id;
        obj.instanceLayer = layer.key;
        obj.instanceIndex = index;
        writeContainerMatrix(layer, index, obj);
        layer.mesh.instanceMatrix.needsUpdate = true;
        if (layer.mesh.instanceColor) {
            layer.mesh.instanceColor.needsUpdate = true;
        }
        return obj;
    }

    function removeContainerInstance(obj) {
        if (!obj.instanceLayer || obj.instanceIndex < 0) {
            return;
        }
        var layer = containerLayers[obj.instanceLayer];
        if (!layer) {
            return;
        }
        hideContainerSlot(layer, obj.instanceIndex);
        layer.idByIndex[obj.instanceIndex] = null;
        layer.freeList.push(obj.instanceIndex);
        layer.mesh.instanceMatrix.needsUpdate = true;
        obj.instanceLayer = null;
        obj.instanceIndex = -1;
        obj.bounds.makeEmpty();
    }

    function moveContainerInstance(obj) {
        if (!obj.instanceLayer || obj.instanceIndex < 0) {
            addContainerInstance(obj);
            return;
        }
        var layer = containerLayers[obj.instanceLayer];
        writeContainerMatrix(layer, obj.instanceIndex, obj);
        layer.mesh.instanceMatrix.needsUpdate = true;
        if (layer.mesh.instanceColor) {
            layer.mesh.instanceColor.needsUpdate = true;
        }
    }

    function buildContainerLayers() {
        var counts = {};
        var oi;
        for (oi = 0; oi < objects.length; oi++) {
            var obj = objects[oi];
            if (!obj || !isContainerKind(obj.kind)) {
                continue;
            }
            var key = layerKey(obj.kind, obj.blockId);
            counts[key] = (counts[key] || 0) + 1;
        }
        for (var ck in counts) {
            if (!Object.prototype.hasOwnProperty.call(counts, ck)) {
                continue;
            }
            var parts = ck.split('|');
            ensureContainerLayer(parts[0], parts[1] === '_' ? '' : parts[1], counts[ck] + 64);
        }
        var built = 0;
        for (oi = 0; oi < objects.length; oi++) {
            obj = objects[oi];
            if (!obj || !isContainerKind(obj.kind)) {
                continue;
            }
            addContainerInstance(obj);
            built++;
        }
        for (ck in containerLayers) {
            if (!Object.prototype.hasOwnProperty.call(containerLayers, ck)) {
                continue;
            }
            var builtLayer = containerLayers[ck];
            builtLayer.mesh.instanceMatrix.needsUpdate = true;
            if (builtLayer.mesh.instanceColor) {
                builtLayer.mesh.instanceColor.needsUpdate = true;
            }
            // На части WebKit computeBoundingSphere у InstancedMesh ломается —
            // оставляем frustumCulled=false и явный большой радиус.
            if (builtLayer.mesh.boundingSphere) {
                builtLayer.mesh.boundingSphere.center.set(
                    site.width / 2, 20, site.length / 2);
                builtLayer.mesh.boundingSphere.radius =
                    Math.max(site.width, site.length);
            }
        }
        return built;
    }

    var renderedEquipment = 0;
    var renderedContainers = 0;
    if (useInstancedContainers) {
        var equipmentIndexes = [];
        for (oi = 0; oi < objects.length; oi++) {
            if (!isContainerKind(objects[oi].kind)) {
                equipmentIndexes.push(oi);
            }
        }
        renderedEquipment = buildObjectBatches(equipmentIndexes, true, { skipContainers: true });
        try {
            renderedContainers = buildContainerLayers();
            if (!renderedContainers) {
                useInstancedContainers = false;
                containerRenderMode = 'batched-fallback';
            }
        } catch (instancedError) {
            useInstancedContainers = false;
            containerRenderMode = 'batched-fallback';
            if (window.console && typeof window.console.error === 'function') {
                window.console.error(instancedError);
            }
        }
    }
    if (!useInstancedContainers) {
        var nonContainerIndexes = [];
        var containerIndexes = [];
        for (oi = 0; oi < objects.length; oi++) {
            if (isContainerKind(objects[oi].kind)) {
                containerIndexes.push(oi);
            } else {
                nonContainerIndexes.push(oi);
            }
        }
        // Сначала техника с рёбрами, затем контейнеры без рёбер.
        renderedEquipment = buildObjectBatches(nonContainerIndexes, true, { skipContainers: true });
        renderedContainers = buildObjectBatches(containerIndexes, true, { skipEdges: true });
    }
    var renderedObjects = renderedEquipment + renderedContainers;
    var nowEnd = (typeof performance !== 'undefined' && performance.now)
        ? performance.now() : Date.now();
    buildMs = Math.round(nowEnd - buildStartedAt);

    function applyLook() {
        var cosP = Math.cos(pitch);
        var dirX = Math.sin(yaw) * cosP;
        var dirY = Math.sin(pitch);
        var dirZ = Math.cos(yaw) * cosP;
        camera.up.set(0, 1, 0);
        camera.lookAt(
            camera.position.x + dirX,
            camera.position.y + dirY,
            camera.position.z + dirZ
        );
    }

    function lookAtPoint(tx, ty, tz) {
        var dx = tx - camera.position.x;
        var dy = ty - camera.position.y;
        var dz = tz - camera.position.z;
        yaw = Math.atan2(dx, dz);
        pitch = Math.atan2(dy, Math.sqrt(dx * dx + dz * dz));
        applyLook();
    }

    function resetCamera() {

        camera.position.set(site.width * 0.05, Math.max(30 * 2.591, maxSize * 0.3), site.length * 0.05);
        camera.fov = 60;
        camera.updateProjectionMatrix();
        lookAtPoint(site.width / 2, 0, site.length / 2);
        needsRender = true;
    }

    function selectObject(obj) {
        clearSelection();
        if (!obj || obj.bounds.isEmpty()) {
            updateStatus();
            return;

    function refreshSelectedObject() {

            clearSelection();

    function addObject(source) {
        var obj = normalizeObject(source);
        if (!obj) {
            throw new Error('Некорректные обязательные поля объекта.');
        }
        if (objectById[obj.id]) {
            throw new Error('Объект с id "' + obj.id + '" уже существует.');
        }
        if (!objectFactories[obj.kind] && !isContainerKind(obj.kind)) {
            throw new Error('Неподдерживаемый kind "' + obj.kind + '".');
        }
        obj.objectIndex = objects.length;
        objects.push(obj);
        objectById[obj.id] = obj;
        if (isContainerKind(obj.kind) && useInstancedContainers) {
            addContainerInstance(obj);
            renderedContainers++;
        } else if (!buildObjectBatches([obj.objectIndex], false, {
            skipEdges: isContainerKind(obj.kind)
        })) {
            objects[obj.objectIndex] = null;
            delete objectById[obj.id];
            throw new Error('Не удалось построить геометрию объекта.');
        } else if (isContainerKind(obj.kind)) {
            renderedContainers++;
        } else {
            renderedEquipment++;
        }
        renderedObjects = renderedEquipment + renderedContainers;
        needsRender = true;
        updateStatus();
        return obj;
    }

    function removeObject(id) {
        id = String(id == null ? '' : id);
        var obj = objectById[id];
        if (!obj) {
            throw new Error('Объект с id "' + id + '" не найден.');
        }
        var objectIndex = obj.objectIndex;
        var wasSelected = selected === obj;
        if (isContainerKind(obj.kind) && useInstancedContainers) {
            removeContainerInstance(obj);
            renderedContainers--;
        } else {
            if (isContainerKind(obj.kind)) {
                renderedContainers--;
            } else {
                renderedEquipment--;
            }
            objects[objectIndex] = null;
            delete objectById[id];
            rebuildBatchesForObject(objectIndex);
            delete objectBatchesByIndex[objectIndex];
            renderedObjects = renderedEquipment + renderedContainers;
            if (wasSelected) {
                clearSelection();
            }
            needsRender = true;
            updateStatus();
            return true;
        }
        objects[objectIndex] = null;
        delete objectById[id];
        renderedObjects = renderedEquipment + renderedContainers;
        if (wasSelected) {
            clearSelection();
        }
        needsRender = true;
        updateStatus();
        return true;
    }

    function moveObject(id, x, y, z) {
        id = String(id == null ? '' : id);
        var obj = objectById[id];
        x = Number(x);
        y = Number(y);
        z = Number(z);
        if (!obj) {
            throw new Error('Объект с id "' + id + '" не найден.');
        }
        if (!isFinite(x) || !isFinite(y) || !isFinite(z)) {
            throw new Error('Координаты должны быть числами.');
        }
        obj.x = x;
        obj.y = y;
        obj.z = z;
        obj.source.x = x;
        obj.source.y = y;
        obj.source.z = z;
        if (isContainerKind(obj.kind) && useInstancedContainers) {
            moveContainerInstance(obj);
        } else {
            rebuildBatchesForObject(obj.objectIndex);
        }
        if (selected === obj) {
            refreshSelectedObject();
        }
        needsRender = true;
        updateStatus();
        return obj;
    }

    function focusObject(id) {
        id = String(id == null ? '' : id);
        var obj = objectById[id];
        if (!obj) {
            throw new Error('Объект с id "' + id + '" не найден.');
        }
        selectObject(obj);
        var center = new THREE.Vector3();
        obj.bounds.getCenter(center);
        camera.position.set(center.x - 40, center.y + 35, center.z - 40);
        lookAtPoint(center.x, center.y, center.z);
        needsRender = true;
        return obj;
    }

    function applyDelta(delta) {
        delta = delta || {};
        var started = (typeof performance !== 'undefined' && performance.now)
            ? performance.now() : Date.now();
        var moves = delta.moves instanceof Array ? delta.moves : [];
        var upserts = delta.upserts instanceof Array ? delta.upserts : [];
        var removed = delta.removed instanceof Array ? delta.removed : [];
        var mi;
        for (mi = 0; mi < moves.length; mi++) {
            var move = moves[mi] || {};
            moveObject(move.id, move.x, move.y, move.z);
        }
        for (mi = 0; mi < upserts.length; mi++) {
            var item = upserts[mi] || {};
            if (objectById[String(item.id)]) {
                var existing = objectById[String(item.id)];
                existing.x = Number(item.x);
                existing.y = Number(item.y);
                existing.z = Number(item.z);
                if (item.rotation != null) {
                    existing.rotation = numberOr(item.rotation, existing.rotation);
                }
                if (item.color) {
                    existing.color = String(item.color);
                }
                if (isContainerKind(existing.kind) && useInstancedContainers) {
                    moveContainerInstance(existing);
                } else {
                    rebuildBatchesForObject(existing.objectIndex);
                }
            } else {
                addObject(item);
            }
        }
        for (mi = 0; mi < removed.length; mi++) {
            removeObject(removed[mi]);
        }
        var ended = (typeof performance !== 'undefined' && performance.now)
            ? performance.now() : Date.now();
        lastDeltaMs = Math.round(ended - started);
        needsRender = true;
        updateStatus();
        return { moves: moves.length, upserts: upserts.length, removed: removed.length, ms: lastDeltaMs };
    }

    function applyCommand(command) {
        command = command || {};
        if (command.op === 'add') {
            return addObject(command.object);
        }
        if (command.op === 'remove') {
            return removeObject(command.id);
        }
        if (command.op === 'move') {
            return moveObject(command.id, command.x, command.y, command.z);
        }
        if (command.op === 'applyDelta') {
            return applyDelta(command.delta);
        }
        if (command.op === 'focus') {
            return focusObject(command.id);
        }
        throw new Error('Неизвестная операция "' + String(command.op || '') + '".');
    }

    function processBridgeCommand(event) {
        if (event) {
            if (event.preventDefault) {
                event.preventDefault();
            }
            if (event.stopPropagation) {
                event.stopPropagation();
            }
        }
        if (!commandBridge) {
            return false;
        }
        var sequence = commandBridge.getAttribute('data-seq') || '';
        if (!sequence || sequence === processBridgeCommand.lastSequence) {
            return false;
        }
        processBridgeCommand.lastSequence = sequence;
        try {
            var payload = commandBridge.getAttribute('data-payload') || '';
            applyCommand(JSON.parse(payload));
            commandBridge.setAttribute('data-error', '');
        } catch (commandError) {
            commandBridge.setAttribute('data-error', commandError.message || String(commandError));
            if (window.console && typeof window.console.error === 'function') {
                window.console.error(commandError);
            }
        }
        commandBridge.setAttribute('data-ack', sequence);
        return false;
    }
    processBridgeCommand.lastSequence = '';

        }
        commandBridge.setAttribute('data-ready', 'true');
    }

    function findObjectIndex(ranges, faceIndex) {
        var low = 0;
        var high = ranges.length - 1;
        while (low <= high) {
            var middle = (low + high) >> 1;
            var range = ranges[middle];
            if (faceIndex < range.startFace) {
                high = middle - 1;
            } else if (faceIndex >= range.endFace) {
                low = middle + 1;
            } else {
                return range.objectIndex;
            }
        }
        return -1;
    }

    var raycaster = new THREE.Raycaster();
    var pointer = new THREE.Vector2();

    function pickAt(clientX, clientY) {
        var rect = renderer.domElement.getBoundingClientRect();
        if (!rect.width || !rect.height) {
            return;
        }
        pointer.x = ((clientX - rect.left) / rect.width) * 2 - 1;
        pointer.y = -((clientY - rect.top) / rect.height) * 2 + 1;
        raycaster.setFromCamera(pointer, camera);
        var pickList = renderMeshes.concat(containerPickMeshes);
        var hits = raycaster.intersectObjects(pickList, false);
        if (!hits.length) {
            clearSelection();
            updateStatus();
            return;
        }
        var hit = hits[0];
        if (hit.object.isInstancedMesh && hit.instanceId != null) {
            var layer = containerLayers[hit.object.userData.layerKey];
            if (layer) {
                var pickedId = layer.idByIndex[hit.instanceId];
                if (pickedId && objectById[pickedId]) {
                    selectObject(objectById[pickedId]);
                    return;
                }
            }
        }
        var objectIndex = findObjectIndex(hit.object.userData.ranges || [], hit.faceIndex);
        if (objectIndex >= 0) {
            selectObject(objects[objectIndex]);
        }
    }

    function resize() {
        var width = viewport.clientWidth || window.innerWidth || 800;
        var height = viewport.clientHeight || window.innerHeight || 600;
        if (width < 1 || height < 1) {
            return;
        }
        var pixelCount = width * height;
        renderScale = pixelCount > MAX_RENDER_PIXELS
            ? Math.sqrt(MAX_RENDER_PIXELS / pixelCount)
            : 1;
        var renderWidth = Math.max(1, Math.round(width * renderScale));
        var renderHeight = Math.max(1, Math.round(height * renderScale));
        camera.aspect = width / height;
        camera.updateProjectionMatrix();
        renderer.setSize(renderWidth, renderHeight, false);
        renderer.domElement.style.width = width + 'px';
        renderer.domElement.style.height = height + 'px';
        needsRender = true;
        updateStatus();
    }
    resize();
    window.addEventListener('resize', resize);
    if (typeof ResizeObserver !== 'undefined') {
        new ResizeObserver(resize).observe(viewport);
    }

    // --- Клавиатура в WebKit ПолеHTMLДокумента ---
    var keyTrap = document.getElementById('key-trap');

    function resolveKeyCode(e) {
        if (e.code) { return e.code; }
        var key = (e.key || '').toLowerCase();
        if (key === 'w' || key === 'ц') { return 'KeyW'; }
        if (key === 'a' || key === 'ф') { return 'KeyA'; }
        if (key === 's' || key === 'ы') { return 'KeyS'; }
        if (key === 'd' || key === 'в') { return 'KeyD'; }
        if (key === 'q' || key === 'й') { return 'KeyQ'; }
        if (key === 'e' || key === 'у') { return 'KeyE'; }
        if (key === ' ') { return 'Space'; }
        if (key === 'arrowup') { return 'ArrowUp'; }
        if (key === 'arrowdown') { return 'ArrowDown'; }
        if (key === 'arrowleft') { return 'ArrowLeft'; }
        if (key === 'arrowright') { return 'ArrowRight'; }
        if (key === 'control') { return 'ControlLeft'; }
        var code = e.keyCode || e.which || 0;
        var map = {
            87: 'KeyW', 65: 'KeyA', 83: 'KeyS', 68: 'KeyD',
            81: 'KeyQ', 69: 'KeyE', 32: 'Space',
            38: 'ArrowUp', 40: 'ArrowDown', 37: 'ArrowLeft',
            39: 'ArrowRight', 17: 'ControlLeft'
        };
        return map[code] || '';
    }

    function onKeyDown(e) {
        var code = resolveKeyCode(e);
        if (!code) { return; }
        keys[code] = true;
        keyLeaseUntil[code] = new Date().getTime() + KEY_LEASE_MS;
        if (e.preventDefault) { e.preventDefault(); }
        return false;
    }

    function onKeyUp(e) {
        var code = resolveKeyCode(e);
        if (!code) { return; }
        keys[code] = false;
        keyLeaseUntil[code] = 0;
        if (e.preventDefault) { e.preventDefault(); }
        return false;
    }

    function clearInputState() {
        var code;
        for (code in keys) {
            if (Object.prototype.hasOwnProperty.call(keys, code)) {
                keys[code] = false;
                keyLeaseUntil[code] = 0;
            }
        }
        looking = false;
        lookLast = null;
        pointerDown = null;
        viewport.classList.remove('looking');
    }

    function isKeyPressed(code, now) {
        if (!keys[code]) {
            return false;
        }
        if (!keyLeaseUntil[code] || keyLeaseUntil[code] < now) {
            keys[code] = false;
            keyLeaseUntil[code] = 0;
            return false;
        }
        return true;
    }

    function focusKeyTrap() {
        if (!keyTrap) { return; }
        try {
            // При возврате фокуса из 1С старое состояние клавиш недостоверно.
            if (document.activeElement && document.activeElement !== keyTrap) {
                clearInputState();
            }
            keyTrap.focus();
            setTimeout(function () {
                try { keyTrap.focus(); } catch (ignore2) {}
            }, 0);
        } catch (ignore) {}
    }

    function bindKeys(target) {
        if (!target) { return; }
        if (target.addEventListener) {
            target.addEventListener('keydown', onKeyDown, false);
            target.addEventListener('keyup', onKeyUp, false);
        } else if (target.attachEvent) {
            target.attachEvent('onkeydown', onKeyDown);
            target.attachEvent('onkeyup', onKeyUp);
        }
    }

    bindKeys(window);
    bindKeys(document);
    bindKeys(document.body);
    bindKeys(keyTrap);
    bindKeys(renderer.domElement);
    // Capture получает keyup раньше элементов документа и снижает шанс его потери.
    if (window.addEventListener) {
        window.addEventListener('keyup', onKeyUp, true);
        window.addEventListener('blur', clearInputState, false);
        window.addEventListener('focus', clearInputState, false);
        window.addEventListener('pagehide', clearInputState, false);
    } else if (window.attachEvent) {
        window.attachEvent('onblur', clearInputState);
        window.attachEvent('onfocus', clearInputState);
    }
    if (document.addEventListener) {
        document.addEventListener('visibilitychange', function () {
            if (document.hidden !== false) {
                clearInputState();
            }
        }, false);
    }
    if (keyTrap) {
        keyTrap.onkeydown = onKeyDown;
        keyTrap.onkeyup = onKeyUp;
        keyTrap.oninput = function () { keyTrap.value = ''; };
    }

    renderer.domElement.addEventListener('mousedown', function (e) {
        if (e.button !== 0) { return; }
        looking = true;
        lookLast = { x: e.clientX, y: e.clientY };
        pointerDown = { x: e.clientX, y: e.clientY, moved: false };
        viewport.classList.add('looking');
        focusKeyTrap();
        e.preventDefault();
    });

    window.addEventListener('mousemove', function (e) {
        if (!looking || !lookLast) { return; }
        var dx = e.clientX - lookLast.x;
        var dy = e.clientY - lookLast.y;
        lookLast = { x: e.clientX, y: e.clientY };
        if (pointerDown) {
            var totalX = e.clientX - pointerDown.x;
            var totalY = e.clientY - pointerDown.y;
            if (totalX * totalX + totalY * totalY > DRAG_THRESHOLD * DRAG_THRESHOLD) {
                pointerDown.moved = true;
            }
        }
        yaw -= dx * 0.005;
        pitch -= dy * 0.005;
        var limit = Math.PI / 2 - 0.05;
        pitch = Math.max(-limit, Math.min(limit, pitch));
        applyLook();
        needsRender = true;
    });

    window.addEventListener('mouseup', function (e) {
        if (e.button !== 0) { return; }
        var wasClick = pointerDown && !pointerDown.moved;
        var clickX = e.clientX;
        var clickY = e.clientY;
        looking = false;
        lookLast = null;
        pointerDown = null;
        viewport.classList.remove('looking');
        focusKeyTrap();
        if (wasClick) { pickAt(clickX, clickY); }
    });

    renderer.domElement.addEventListener('wheel', function (e) {
        e.preventDefault();
        var next = camera.fov + (e.deltaY > 0 ? 3 : -3);
        next = Math.max(20, Math.min(90, next));
        if (next !== camera.fov) {
            camera.fov = next;
            camera.updateProjectionMatrix();
            needsRender = true;
            updateStatus();
        }
    }, { passive: false });

    var btnReset = document.getElementById('btn-reset');
    if (btnReset) {
        btnReset.addEventListener('click', function () {
            resetCamera();
            focusKeyTrap();
        });
    }

    function moveCamera(dt) {
        var distance = speedMetersPerSecond * dt;
        var moved = false;
        var now = new Date().getTime();
        var cosP = Math.cos(pitch);
        var forward = new THREE.Vector3(
            Math.sin(yaw) * cosP, Math.sin(pitch), Math.cos(yaw) * cosP
        ).normalize();
        var right = new THREE.Vector3(
            Math.sin(yaw + Math.PI / 2), 0, Math.cos(yaw + Math.PI / 2)
        ).normalize();
        if (isKeyPressed('KeyW', now) || isKeyPressed('ArrowUp', now)) {
            camera.position.addScaledVector(forward, distance);
            moved = true;
        }
        if (isKeyPressed('KeyS', now) || isKeyPressed('ArrowDown', now)) {
            camera.position.addScaledVector(forward, -distance);
            moved = true;
        }
        // A/D и стрелки оставлены в выбранной пользователем обратной схеме.
        if (isKeyPressed('KeyA', now) || isKeyPressed('ArrowLeft', now)) {
            camera.position.addScaledVector(right, distance);
            moved = true;
        }
        if (isKeyPressed('KeyD', now) || isKeyPressed('ArrowRight', now)) {
            camera.position.addScaledVector(right, -distance);
            moved = true;
        }
        if (isKeyPressed('KeyE', now) || isKeyPressed('Space', now)) {
            camera.position.y += distance;
            moved = true;
        }
        if (isKeyPressed('KeyQ', now) ||
                isKeyPressed('ControlLeft', now) ||
                isKeyPressed('ControlRight', now)) {
            camera.position.y -= distance;
            moved = true;
        }
        if (moved) {
            applyLook();
        }
        return moved;
    }

    updateStatus();
    function animate() {
        requestAnimationFrame(animate);
        var dt = Math.min(clock.getDelta(), 0.1);
        var moved = moveCamera(dt);
        if (needsRender || moved) {
            renderer.render(scene, camera);
            needsRender = false;
            if (moved && dt > 0) {
                fpsAccum += 1 / dt;
                fpsFrames++;
                if (fpsFrames >= 20) {
                    fpsValue = Math.round(fpsAccum / fpsFrames);
                    fpsAccum = 0;
                    fpsFrames = 0;
                    updateStatus();
                }
            }
        }
    }
    animate();
}

if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initWarehouse3D);
} else {
    initWarehouse3D();
}
